cmake_minimum_required(VERSION 4.4)
project(simd_api_examples LANGUAGES CXX)
find_package(simd CONFIG REQUIRED COMPONENTS simd)
enable_testing()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)$")
  set(api_default_profile NEON)
else()
  set(api_default_profile AVX2)
endif()
set(API_PROFILES "${api_default_profile}" CACHE STRING "Architectures admitted for example execution")
foreach(profile IN LISTS API_PROFILES)
  if(NOT profile MATCHES "^(AVX2|AVX512|NEON)$")
    message(FATAL_ERROR "Unsupported example profile ${profile}")
  endif()
  string(TOLOWER "${profile}" lower)
  add_executable(api_${lower} api.cc)
  target_link_libraries(api_${lower} PRIVATE simd::simd simd::common simd::${lower})
  simd_target_profile(api_${lower} ${profile})
  target_compile_definitions(api_${lower} PRIVATE API_${profile}=1)
  set_target_properties(api_${lower} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  add_test(NAME api.${lower} COMMAND api_${lower})
endforeach()
