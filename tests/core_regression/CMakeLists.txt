set(NATIVE_RAW_SCALEB_FTZ_PROFILE "before-rounding" CACHE STRING
  "Previously captured native raw scaling tininess: before-rounding or after-rounding")

if(NOT NATIVE_RAW_SCALEB_FTZ_PROFILE MATCHES "^(before-rounding|after-rounding)$")
  message(FATAL_ERROR "Select a captured raw scaling FTZ profile")

foreach(suite IN ITEMS integer mask bit_bridge scaleb wide_exp)
  add_executable(native_test_${suite} ${suite}.cc)
  target_link_libraries(native_test_${suite} PRIVATE native::native)
  target_include_directories(native_test_${suite} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(native_test_${suite} PRIVATE /W4 /WX /wd4127 /wd4702)
    set_property(TARGET native_test_${suite} APPEND_STRING PROPERTY COMPILE_FLAGS " /bigobj")
    # Guarded/instrumented wide register fixtures can require a large frame.
    target_link_options(native_test_${suite} PRIVATE /STACK:16777216)
  else()
    target_compile_options(native_test_${suite} PRIVATE -Wall -Wextra -Werror)
    if(WIN32)
      target_link_options(native_test_${suite} PRIVATE -Xlinker /STACK:16777216)
    endif()
  endif()
endforeach()
