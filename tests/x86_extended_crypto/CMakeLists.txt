cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_x86_extended_crypto_check LANGUAGES CXX)
  set(NATIVE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../.. native EXCLUDE_FROM_ALL)
  enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(crypto_objdump NAMES llvm-objdump HINTS "${compiler_directory}" REQUIRED)
function(crypto_options target)
  set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  foreach(option IN ITEMS -O2 -fconstexpr-steps=8000000)
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
      target_compile_options(${target} PRIVATE "/clang:${option}")
    else()
      target_compile_options(${target} PRIVATE "${option}")
    endif()
  endforeach()
endfunction()
foreach(mode IN ITEMS raw public)
  add_library(native_test_x86_extended_crypto_${mode} OBJECT codegen.cc)
  target_link_libraries(native_test_x86_extended_crypto_${mode} PRIVATE native::native)
  crypto_options(native_test_x86_extended_crypto_${mode})
  if(mode STREQUAL "raw")
    target_compile_definitions(native_test_x86_extended_crypto_${mode} PRIVATE NATIVE_CRYPTO_RAW=1)
  endif()
endforeach()
add_test(NAME native.x86.extended_crypto.codegen COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../x86_codegen_pairs.py" --objdump "${crypto_objdump}"
  --raw "$<TARGET_OBJECTS:native_test_x86_extended_crypto_raw>"
  --public "$<TARGET_OBJECTS:native_test_x86_extended_crypto_public>" --expected 13
  --output "${CMAKE_CURRENT_BINARY_DIR}/paired-codegen.txt")

foreach(family IN ITEMS sha512 sm3 sm4)
  add_test(NAME native.x86.extended_crypto.runtime_${family} COMMAND native_test_x86_extended_crypto_runtime ${family})
  set_tests_properties(native.x86.extended_crypto.runtime_${family} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
add_executable(native_test_x86_extended_crypto_metadata metadata.cc)
target_link_libraries(native_test_x86_extended_crypto_metadata PRIVATE native::headers)
target_compile_features(native_test_x86_extended_crypto_metadata PRIVATE cxx_std_26)
set_target_properties(native_test_x86_extended_crypto_metadata PROPERTIES CXX_SCAN_FOR_MODULES OFF)
add_test(NAME native.x86.extended_crypto.metadata COMMAND native_test_x86_extended_crypto_metadata)
foreach(family RANGE 0 2)
  set(last_kind 4)
  if(family EQUAL 1)
    set(last_kind 5)
  endif()
  foreach(kind RANGE 0 ${last_kind})
    set(target native_test_x86_extended_crypto_negative_${family}_${kind})
    if(kind EQUAL 0)
      add_library(${target} OBJECT negative.cc)
    else()
      add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
    endif()
    target_link_libraries(${target} PRIVATE native::native)
    crypto_options(${target})
    target_compile_definitions(${target} PRIVATE NATIVE_CRYPTO_FAMILY=${family} NATIVE_CRYPTO_NEGATIVE=${kind})
    if(kind GREATER 0)
      set(diagnostic "deleted function|constraints not satisfied")
      if(kind EQUAL 1)
        set(diagnostic "call to consteval function.*is not a constant expression")
      elseif(kind EQUAL 2)
        set(diagnostic "requires target feature")
      endif()
      add_test(NAME native.x86.extended_crypto.reject_${family}_${kind} COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../source_targets/check_negative.py"
        --cmake "${CMAKE_COMMAND}" --build "${CMAKE_BINARY_DIR}" --target "${target}"
        --config "$<CONFIG>" --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${family}-${kind}.log"
        --diagnostic "${diagnostic}")
      set_tests_properties(native.x86.extended_crypto.reject_${family}_${kind} PROPERTIES RUN_SERIAL TRUE)
    endif()
  endforeach()
endforeach()
