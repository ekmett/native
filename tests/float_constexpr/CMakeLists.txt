cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_float_constexpr_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
endif()
set(modes modules)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  list(APPEND modes headers)
endif()
foreach(mode IN LISTS modes)
  set(target native_test_float_constexpr_${mode})
  add_executable(${target} main.cc)
  target_link_libraries(${target} PRIVATE native::native)
  set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(mode STREQUAL "headers")
    target_compile_definitions(${target} PRIVATE NATIVE_FLOAT_HEADERS=1)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -ffp-model=strict)
  endif()
  add_test(NAME native.float.constexpr_${mode} COMMAND ${target})
  set_tests_properties(native.float.constexpr_${mode} PROPERTIES LABELS "host;math;constexpr;property")
endforeach()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  return()
endif()
add_library(native_test_float_constexpr_codegen OBJECT codegen.cc)
target_link_libraries(native_test_float_constexpr_codegen PRIVATE native::native)
set_target_properties(native_test_float_constexpr_codegen PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
if(MSVC)
  target_compile_options(native_test_float_constexpr_codegen PRIVATE /O2 /clang:-fomit-frame-pointer)
else()
  target_compile_options(native_test_float_constexpr_codegen PRIVATE -O2 -fomit-frame-pointer)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(float_objdump NAMES llvm-objdump HINTS "${compiler_dir}" REQUIRED)
add_test(NAME native.float.constexpr_codegen COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/check_codegen.py" --objdump "${float_objdump}"
  --output "${CMAKE_CURRENT_BINARY_DIR}/float-$<CONFIG>.disassembly"
  $<TARGET_OBJECTS:native_test_float_constexpr_codegen> COMMAND_EXPAND_LISTS)
set_tests_properties(native.float.constexpr_codegen PROPERTIES LABELS "host;math;constexpr;codegen")
add_test(NAME native.float.codegen_parser COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/test_codegen.py")
set_tests_properties(native.float.codegen_parser PROPERTIES LABELS "host;math;codegen;python")
