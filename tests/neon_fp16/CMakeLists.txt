cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_neon_fp16_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
  set(neon_fp16_standalone TRUE)
endif()
if(NOT TARGET native::neon_fp16)
  if(neon_fp16_standalone)
    message(FATAL_ERROR "This fixture requires a package for a host supporting NEON_FP16.")
  endif()
  return()
endif()
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT "NEON_FP16" IN_LIST NATIVE_PROFILES)
  return()
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(native_neon_fp16_kernel OBJECT kernel.cc)
target_link_libraries(native_neon_fp16_kernel PRIVATE native::minimal native::neon_fp16)
native_target_profile(native_neon_fp16_kernel NEON_FP16)
target_include_directories(native_neon_fp16_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
set_target_properties(native_neon_fp16_kernel PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_library(native_neon_fp16_omnibus OBJECT omnibus.cc)
target_link_libraries(native_neon_fp16_omnibus PRIVATE native::native native::minimal)
native_target_omnibus(native_neon_fp16_omnibus)
target_precompile_headers(native_neon_fp16_omnibus PRIVATE <array> <concepts> <cstddef> <type_traits>)
add_executable(native_test_neon_fp16 main.cc $<TARGET_OBJECTS:native_neon_fp16_kernel>)
target_link_libraries(native_test_neon_fp16 PRIVATE native::minimal native::neon_fp16)
set_target_properties(native_test_neon_fp16 PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_dependencies(native_test_neon_fp16 native_neon_fp16_omnibus)
if(COMMAND native_link_clang_cl_asan)
  native_link_clang_cl_asan(native_test_neon_fp16)
endif()
if(TARGET native::neon)
  add_library(native_neon_fp16_identity OBJECT identity.cc)
  target_link_libraries(native_neon_fp16_identity PRIVATE native::minimal native::neon native::neon_fp16)
  native_target_profile(native_neon_fp16_identity NEON_FP16)
  add_dependencies(native_test_neon_fp16 native_neon_fp16_identity)
endif()
foreach(mode IN ITEMS none admission native)
  add_test(NAME native.neon_fp16.${mode} COMMAND native_test_neon_fp16 ${mode})
  set_tests_properties(native.neon_fp16.${mode} PROPERTIES SKIP_RETURN_CODE 77
    LABELS "host;modules;neon_fp16;exact" TIMEOUT 180)
endforeach()
add_test(NAME native.neon_fp16.baseline_features COMMAND "${CMAKE_COMMAND}"
  "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/main.cc"
  -P "${CMAKE_CURRENT_SOURCE_DIR}/baseline_features.cmake")
set_tests_properties(native.neon_fp16.baseline_features PROPERTIES LABELS "host;modules;neon_fp16;profiles")
get_filename_component(neon_fp16_compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(NATIVE_NEON_FP16_OBJDUMP NAMES llvm-objdump HINTS "${neon_fp16_compiler_directory}")
get_target_property(neon_fp16_usage_options native::minimal INTERFACE_COMPILE_OPTIONS)
# Sanitizer instrumentation deliberately introduces calls; inspect ordinary objects.
if(NATIVE_NEON_FP16_OBJDUMP AND NOT neon_fp16_usage_options MATCHES "sanitize=address")
  add_test(NAME native.neon_fp16.codegen COMMAND "${CMAKE_COMMAND}"
    "-DOBJDUMP=${NATIVE_NEON_FP16_OBJDUMP}" "-DOBJECT=$<TARGET_OBJECTS:native_neon_fp16_kernel>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/codegen.cmake")
  set_tests_properties(native.neon_fp16.codegen PROPERTIES LABELS "host;modules;neon_fp16;codegen")
endif()
if(neon_fp16_standalone AND CMAKE_GENERATOR STREQUAL "Ninja")
  add_test(NAME native.neon_fp16.shared_bmi COMMAND "${CMAKE_COMMAND}"
    "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DMODULES=native.scalar;native.wide;native.simd;native.numerics;native.arm.features;native.features;native"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/../profile_package/check_bmi.cmake")
  set_tests_properties(native.neon_fp16.shared_bmi PROPERTIES LABELS "host;modules;neon_fp16;profiles")
endif()
