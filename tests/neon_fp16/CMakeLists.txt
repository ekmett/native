cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(simd_neon_fp16_consumer LANGUAGES CXX)
  find_package(simd CONFIG REQUIRED COMPONENTS simd)
  enable_testing()
  set(neon_fp16_standalone TRUE)
endif()
if(NOT TARGET simd::neon_fp16)
  if(neon_fp16_standalone)
    message(FATAL_ERROR "This fixture requires a package for a host supporting NEON_FP16.")
  endif()
  return()
endif()
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT "NEON_FP16" IN_LIST SIMD_PROFILES)
  return()
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(simd_neon_fp16_kernel OBJECT kernel.cc)
target_link_libraries(simd_neon_fp16_kernel PRIVATE simd::minimal simd::neon_fp16)
simd_target_profile(simd_neon_fp16_kernel NEON_FP16)
target_include_directories(simd_neon_fp16_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
set_target_properties(simd_neon_fp16_kernel PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_library(simd_neon_fp16_omnibus OBJECT omnibus.cc)
target_link_libraries(simd_neon_fp16_omnibus PRIVATE simd::simd simd::minimal)
simd_target_omnibus(simd_neon_fp16_omnibus)
target_precompile_headers(simd_neon_fp16_omnibus PRIVATE <array> <concepts> <cstddef> <type_traits>)
add_executable(simd_test_neon_fp16 main.cc $<TARGET_OBJECTS:simd_neon_fp16_kernel>)
target_link_libraries(simd_test_neon_fp16 PRIVATE simd::minimal simd::neon_fp16)
set_target_properties(simd_test_neon_fp16 PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_dependencies(simd_test_neon_fp16 simd_neon_fp16_omnibus)
if(COMMAND simd_link_clang_cl_asan)
  simd_link_clang_cl_asan(simd_test_neon_fp16)
endif()
if(TARGET simd::neon)
  add_library(simd_neon_fp16_identity OBJECT identity.cc)
  target_link_libraries(simd_neon_fp16_identity PRIVATE simd::minimal simd::neon simd::neon_fp16)
  simd_target_profile(simd_neon_fp16_identity NEON_FP16)
  add_dependencies(simd_test_neon_fp16 simd_neon_fp16_identity)
endif()
foreach(mode IN ITEMS none admission native)
  add_test(NAME simd.neon_fp16.${mode} COMMAND simd_test_neon_fp16 ${mode})
  set_tests_properties(simd.neon_fp16.${mode} PROPERTIES SKIP_RETURN_CODE 77
    LABELS "host;modules;neon_fp16;exact" TIMEOUT 180)
endforeach()
add_test(NAME simd.neon_fp16.baseline_features COMMAND "${CMAKE_COMMAND}"
  "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/main.cc"
  -P "${CMAKE_CURRENT_SOURCE_DIR}/baseline_features.cmake")
set_tests_properties(simd.neon_fp16.baseline_features PROPERTIES LABELS "host;modules;neon_fp16;profiles")
get_filename_component(neon_fp16_compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(SIMD_NEON_FP16_OBJDUMP NAMES llvm-objdump HINTS "${neon_fp16_compiler_directory}")
get_target_property(neon_fp16_usage_options simd::minimal INTERFACE_COMPILE_OPTIONS)
# Sanitizer instrumentation deliberately introduces calls; inspect ordinary objects.
if(SIMD_NEON_FP16_OBJDUMP AND NOT neon_fp16_usage_options MATCHES "sanitize=address")
  add_test(NAME simd.neon_fp16.codegen COMMAND "${CMAKE_COMMAND}"
    "-DOBJDUMP=${SIMD_NEON_FP16_OBJDUMP}" "-DOBJECT=$<TARGET_OBJECTS:simd_neon_fp16_kernel>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/codegen.cmake")
  set_tests_properties(simd.neon_fp16.codegen PROPERTIES LABELS "host;modules;neon_fp16;codegen")
endif()
if(neon_fp16_standalone AND CMAKE_GENERATOR STREQUAL "Ninja")
  add_test(NAME simd.neon_fp16.shared_bmi COMMAND "${CMAKE_COMMAND}"
    "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DMODULES=simd.scalar;simd.wide;simd.numerics;simd.cpu.arm;simd.cpu;simd"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/../profile_package/check_bmi.cmake")
  set_tests_properties(simd.neon_fp16.shared_bmi PROPERTIES LABELS "host;modules;neon_fp16;profiles")
endif()
