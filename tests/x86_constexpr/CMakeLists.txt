cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_x86_constexpr_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
  return()
endif()

set(interfaces header module hub)
set(index 0)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(interfaces module hub)
  set(index 1)
endif()
foreach(interface IN LISTS interfaces)
  set(target native_test_x86_constexpr_${interface})
  add_executable(${target} main.cc)
  target_compile_definitions(${target} PRIVATE NATIVE_CONSTEXPR_INTERFACE=${index})
  if(interface STREQUAL "header")
    target_link_libraries(${target} PRIVATE native::headers)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON
      CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
  elseif(interface STREQUAL "module")
    target_link_libraries(${target} PRIVATE native::minimal)
  else()
    target_link_libraries(${target} PRIVATE native::native)
  endif()
  if(COMMAND native_link_clang_cl_asan)
    native_link_clang_cl_asan(${target})
  endif()
  set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  add_test(NAME native.x86.constexpr.${interface} COMMAND ${target})
  set_tests_properties(native.x86.constexpr.${interface} PROPERTIES LABELS "host;constexpr;baseline;x86")
  math(EXPR index "${index}+1")
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  return()
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(constexpr_objdump NAMES llvm-objdump HINTS "${compiler_directory}" REQUIRED)
foreach(mode IN ITEMS raw public)
  set(target native_x86_constexpr_codegen_${mode})
  add_library(${target} OBJECT codegen.cc)
  if(mode STREQUAL "raw")
    target_link_libraries(${target} PRIVATE native::headers)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE NATIVE_CONSTEXPR_RAW=1)
    set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES OFF)
  else()
    target_link_libraries(${target} PRIVATE native::minimal)
  endif()
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /O2)
  else()
    target_compile_options(${target} PRIVATE -O2)
  endif()
  set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
endforeach()
add_test(NAME native.x86.constexpr.codegen COMMAND "${Python3_EXECUTABLE}"
  "${PROJECT_SOURCE_DIR}/tests/x86_codegen_pairs.py" --objdump "${constexpr_objdump}"
  --raw "$<TARGET_OBJECTS:native_x86_constexpr_codegen_raw>"
  --public "$<TARGET_OBJECTS:native_x86_constexpr_codegen_public>"
  --output "${CMAKE_CURRENT_BINARY_DIR}/constexpr-codegen-$<CONFIG>.txt")
set_tests_properties(native.x86.constexpr.codegen PROPERTIES LABELS "host;codegen;constexpr;baseline;x86")

foreach(interface IN ITEMS header module)
  foreach(index RANGE 0 58)
    set(target native_x86_constexpr_reject_${interface}_${index})
    add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
    target_compile_definitions(${target} PRIVATE NATIVE_CONSTEXPR_NEGATIVE=${index})
    if(interface STREQUAL "header")
      target_compile_definitions(${target} PRIVATE NATIVE_CONSTEXPR_INTERFACE=0)
      target_link_libraries(${target} PRIVATE native::headers)
      set_target_properties(${target} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON
        CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
    else()
      target_compile_definitions(${target} PRIVATE NATIVE_CONSTEXPR_INTERFACE=1)
      target_link_libraries(${target} PRIVATE native::minimal)
    endif()
    if(index LESS 49)
      set(diagnostic "call to consteval function.*is not a constant expression")
    else()
      set(diagnostic "no viable conversion.*for template parameter 'Arch'")
    endif()
    add_test(NAME native.x86.constexpr.reject_${interface}_${index} COMMAND "${Python3_EXECUTABLE}"
      "${PROJECT_SOURCE_DIR}/tests/source_targets/check_negative.py" --cmake "${CMAKE_COMMAND}"
      --build "${CMAKE_BINARY_DIR}" --target "${target}" --config "$<CONFIG>"
      --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${interface}-${index}-$<CONFIG>.log"
      --diagnostic "${diagnostic}")
    set_tests_properties(native.x86.constexpr.reject_${interface}_${index} PROPERTIES RUN_SERIAL TRUE
      LABELS "host;negative;constexpr;baseline;x86")
  endforeach()
endforeach()
