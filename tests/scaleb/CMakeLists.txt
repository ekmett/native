cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_scaleb_admission LANGUAGES CXX)
  if(NATIVE_SCALEB_SOURCE_PROVIDER)
    set(NATIVE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(../.. native EXCLUDE_FROM_ALL)
  else()
    find_package(native CONFIG REQUIRED COMPONENTS native)
  endif()
  enable_testing()
endif()
set(interfaces modules)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR OR NATIVE_SCALEB_SOURCE_PROVIDER)
  list(APPEND interfaces headers)
endif()
foreach(interface IN LISTS interfaces)
  add_executable(native_test_scaleb_admission_${interface} admission.cc)
  target_link_libraries(native_test_scaleb_admission_${interface} PRIVATE native::native)
  if(interface STREQUAL "headers")
    target_compile_definitions(native_test_scaleb_admission_${interface} PRIVATE NATIVE_SCALEB_HEADERS=1)
  endif()
  add_test(NAME native.scaleb.admission_${interface} COMMAND native_test_scaleb_admission_${interface})
endforeach()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  return()
endif()
add_executable(native_test_scaleb_padding runtime.cc)
target_link_libraries(native_test_scaleb_padding PRIVATE native::native)
add_test(NAME native.scaleb.padding COMMAND native_test_scaleb_padding)
set_tests_properties(native.scaleb.padding PROPERTIES SKIP_RETURN_CODE 77)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(scaleb_objdump NAMES llvm-objdump HINTS "${compiler_directory}" REQUIRED)
foreach(vl RANGE 0 1)
  foreach(mode IN ITEMS raw public)
    set(target native_test_scaleb_${mode}_${vl})
    add_library(${target} OBJECT codegen.cc)
    target_link_libraries(${target} PRIVATE native::native)
    target_compile_definitions(${target} PRIVATE NATIVE_SCALEB_VL=${vl})
    if(mode STREQUAL "raw")
      target_compile_definitions(${target} PRIVATE NATIVE_SCALEB_RAW=1)
    endif()
    set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
      target_compile_options(${target} PRIVATE /clang:-O2)
    else()
      target_compile_options(${target} PRIVATE -O2)
    endif()
  endforeach()
  if(vl)
    set(pairs 18)
  else()
    set(pairs 6)
  endif()
  add_test(NAME native.scaleb.codegen_${vl} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../x86_codegen_pairs.py" --objdump "${scaleb_objdump}"
    --raw "$<TARGET_OBJECTS:native_test_scaleb_raw_${vl}>"
    --public "$<TARGET_OBJECTS:native_test_scaleb_public_${vl}>" --expected ${pairs}
    --output "${CMAKE_CURRENT_BINARY_DIR}/paired-codegen-${vl}.txt")
endforeach()
foreach(kind RANGE 1 4)
  set(target native_test_scaleb_reject_${kind})
  add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
  target_link_libraries(${target} PRIVATE native::native)
  target_compile_definitions(${target} PRIVATE NATIVE_SCALEB_NEGATIVE=${kind})
  set(diagnostic "constraints not satisfied|no matching function")
  if(kind EQUAL 3)
    set(diagnostic "always_inline function 'scaleb' requires target feature")
  endif()
  add_test(NAME native.scaleb.reject_${kind} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../source_targets/check_negative.py"
    --cmake "${CMAKE_COMMAND}" --build "${CMAKE_BINARY_DIR}" --target ${target}
    --config "$<CONFIG>" --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${kind}.log"
    --diagnostic "${diagnostic}")
  set_tests_properties(native.scaleb.reject_${kind} PROPERTIES RUN_SERIAL TRUE)
endforeach()
