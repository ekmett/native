cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_exp_policy_refinement LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native headers)
  enable_testing()
endif()
add_executable(native_test_exp_refinement_metadata metadata.cc)
target_link_libraries(native_test_exp_refinement_metadata PRIVATE native::headers)
target_include_directories(native_test_exp_refinement_metadata PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
target_compile_features(native_test_exp_refinement_metadata PRIVATE cxx_std_20)
set_target_properties(native_test_exp_refinement_metadata PROPERTIES CXX_STANDARD 20)
if(COMMAND native_link_clang_cl_asan)
  native_link_clang_cl_asan(native_test_exp_refinement_metadata)
endif()
add_test(NAME native.exp_refinement.metadata COMMAND native_test_exp_refinement_metadata)
set_tests_properties(native.exp_refinement.metadata PROPERTIES LABELS "host;exp_refinement")

# Native checks retain eleven caller tags across five binary32 implementation policies.
# Metadata above remains host-independent; no ARM code generation is claimed.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
  add_executable(native_test_exp_refinement_import import.cc)
  target_link_libraries(native_test_exp_refinement_import PRIVATE native::native)
  if(COMMAND native_link_clang_cl_asan)
    native_link_clang_cl_asan(native_test_exp_refinement_import)
  endif()
  add_test(NAME native.exp_refinement.import COMMAND native_test_exp_refinement_import)
  set_tests_properties(native.exp_refinement.import PROPERTIES SKIP_RETURN_CODE 77 LABELS "host;exp_refinement")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  get_filename_component(exp_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
  find_program(exp_objdump NAMES llvm-objdump HINTS "${exp_compiler_dir}" REQUIRED)
  add_library(native_exp_refinement_native OBJECT native.cc)
  target_link_libraries(native_exp_refinement_native PRIVATE native::native)
  target_include_directories(native_exp_refinement_native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
  set_target_properties(native_exp_refinement_native PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  add_executable(native_test_exp_refinement_native $<TARGET_OBJECTS:native_exp_refinement_native>)
  target_link_libraries(native_test_exp_refinement_native PRIVATE native::native)
  set_target_properties(native_test_exp_refinement_native PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(COMMAND native_link_clang_cl_asan)
    native_link_clang_cl_asan(native_test_exp_refinement_native)
  endif()
  add_test(NAME native.exp_refinement.native COMMAND native_test_exp_refinement_native)
  set_tests_properties(native.exp_refinement.native PROPERTIES SKIP_RETURN_CODE 77 LABELS "host;exp_refinement" TIMEOUT 120)
  add_test(NAME native.exp_refinement.codegen COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/check_codegen.py" --objdump "${exp_objdump}"
    --output "${CMAKE_CURRENT_BINARY_DIR}/exp-$<CONFIG>.disassembly"
    $<TARGET_OBJECTS:native_exp_refinement_native> COMMAND_EXPAND_LISTS)
  set_tests_properties(native.exp_refinement.codegen PROPERTIES LABELS "host;exp_refinement" TIMEOUT 120)
endif()
