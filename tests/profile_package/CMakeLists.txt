cmake_minimum_required(VERSION 4.4)
project(profile_consumer LANGUAGES CXX)
find_package(simd CONFIG REQUIRED COMPONENTS simd)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()
foreach(profile AVX2 AVX512)
  string(TOLOWER "${profile}" lower)
  add_library(kernel_${lower} OBJECT kernel.cc pair.cc)
  target_link_libraries(kernel_${lower} PRIVATE simd::common simd::${lower})
  simd_target_profile(kernel_${lower} ${profile})
  target_compile_definitions(kernel_${lower} PRIVATE PROFILE_${profile}=1)
endforeach()
add_library(identity OBJECT identity.cc)
target_link_libraries(identity PRIVATE simd::common simd::avx2 simd::avx512)
simd_target_profile(identity AVX512)
add_library(static_string_object OBJECT static_string.cc static_string_peer.cc)
target_link_libraries(static_string_object PRIVATE simd::common)
set_property(TARGET static_string_object PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
add_library(dispatch_object OBJECT dispatch.cc)
target_link_libraries(dispatch_object PRIVATE simd::common)
set_property(TARGET dispatch_object PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
add_executable(dispatch $<TARGET_OBJECTS:dispatch_object> $<TARGET_OBJECTS:kernel_avx2> $<TARGET_OBJECTS:kernel_avx512> $<TARGET_OBJECTS:identity> $<TARGET_OBJECTS:static_string_object>)
target_link_libraries(dispatch PRIVATE simd::avx2 simd::avx512 simd::common)
set_property(TARGET dispatch PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
add_test(NAME imported_profiles COMMAND dispatch)
set_tests_properties(imported_profiles PROPERTIES SKIP_RETURN_CODE 77)

target_include_directories(dispatch_object PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")

add_test(NAME shared_baseline_bmi COMMAND "${CMAKE_COMMAND}"
  "-DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}"
  -P "${CMAKE_CURRENT_SOURCE_DIR}/check_bmi.cmake")
