cmake_minimum_required(VERSION 4.4)
project(simd_memory_portability LANGUAGES CXX)
option(SIMD_MEMORY_ASAN "Instrument the focused memory tests" OFF)
enable_testing()
get_filename_component(simd_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(memory_module STATIC)
target_sources(memory_module PUBLIC FILE_SET CXX_MODULES
  BASE_DIRS "${simd_root}/src/cxx/modules"
  FILES "${simd_root}/src/cxx/modules/simd.memory.ccm")
add_executable(memory_host host.cc attributes.cc)
target_link_libraries(memory_host PRIVATE memory_module)
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  # CMake supplies -TP for .ccm; Clang already infers the module language.
  target_compile_options(memory_module PRIVATE -Wno-unused-command-line-argument)
endif()
set(memory_targets memory_module memory_host)
if(UNIX)
  add_executable(memory_posix posix.cc)
  target_link_libraries(memory_posix PRIVATE memory_module)
  list(APPEND memory_targets memory_posix)
endif()
foreach(target IN LISTS memory_targets)
  target_compile_features(${target} PRIVATE cxx_std_26)
  target_include_directories(${target} PRIVATE "${simd_root}/src/cxx" "${simd_root}/src/shared")
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES ON)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /W4 /WX /EHs-c-)
    target_compile_definitions(${target} PRIVATE _HAS_EXCEPTIONS=0)
    if(SIMD_MEMORY_ASAN)
      if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        message(FATAL_ERROR "This Windows ASan recipe requires clang-cl")
      endif()
      execute_process(COMMAND "${CMAKE_CXX_COMPILER}" --print-runtime-dir
        OUTPUT_VARIABLE runtime_dir OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
      file(TO_CMAKE_PATH "${runtime_dir}" runtime_dir)
      target_compile_options(${target} PRIVATE /fsanitize=address)
      target_link_libraries(${target} PRIVATE "${runtime_dir}/clang_rt.asan_dynamic-x86_64.lib")
      target_link_options(${target} PRIVATE
        "/wholearchive:${runtime_dir}/clang_rt.asan_dynamic_runtime_thunk-x86_64.lib"
        /include:__asan_seh_interceptor /INCREMENTAL:NO)
      add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${runtime_dir}/clang_rt.asan_dynamic-x86_64.dll" "$<TARGET_FILE_DIR:${target}>" VERBATIM)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-exceptions)
    if(SIMD_MEMORY_ASAN)
      target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address)
    endif()
  endif()
  if(NOT target STREQUAL "memory_module")
    add_test(NAME ${target} COMMAND ${target})
  endif()
endforeach()
