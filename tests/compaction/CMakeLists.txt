foreach(profile IN ITEMS SCALAR ${SIMD_PROFILES})
  string(TOLOWER "${profile}" lower)
  if(profile MATCHES "^AVX512")
    set(bits 512)
  elseif(profile STREQUAL "AVX2")
    set(bits 256)
  elseif(profile MATCHES "^NEON")
    set(bits 128)
  else()
    set(bits 0)
  endif()
  foreach(kind IN ITEMS header import)
    set(target simd_compaction_${lower}_${kind})
    add_library(${target}_body OBJECT main.cc)
    target_include_directories(${target}_body PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
    add_executable(${target} driver.cc $<TARGET_OBJECTS:${target}_body>)
    target_compile_definitions(${target} PRIVATE SIMD_TEST_PROFILE=${bits})
    target_link_libraries(${target} PRIVATE simd::common simd::simd)
    set_target_properties(${target} ${target}_body PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
    target_compile_definitions(${target}_body PRIVATE SIMD_TEST_PROFILE=${bits})
    if(profile STREQUAL "AVX512_BF16" OR profile STREQUAL "NEON_BF16")
      target_compile_definitions(${target} PRIVATE SIMD_TEST_BF16=1)
      target_compile_definitions(${target}_body PRIVATE SIMD_TEST_BF16=1)
    endif()
    if(profile STREQUAL "AVX512_FP16")
      target_compile_definitions(${target} PRIVATE SIMD_TEST_AVX512_FP16=1)
      target_compile_definitions(${target}_body PRIVATE SIMD_TEST_AVX512_FP16=1)
    endif()
    if(profile STREQUAL "NEON_FP16")
      target_compile_definitions(${target} PRIVATE SIMD_TEST_FP16=1)
      target_compile_definitions(${target}_body PRIVATE SIMD_TEST_FP16=1)
    endif()
    target_link_libraries(${target}_body PRIVATE simd::common)
    if(kind STREQUAL "import")
      target_compile_definitions(${target}_body PRIVATE SIMD_TEST_IMPORT=1)
    endif()
    if(NOT profile STREQUAL "SCALAR")
      simd_target_profile(${target}_body ${profile})
      target_link_libraries(${target}_body PRIVATE simd::${lower})
    endif()
    simd_link_clang_cl_asan(${target})
    add_test(NAME simd.compaction.${lower}.${kind} COMMAND ${target})
    set_tests_properties(simd.compaction.${lower}.${kind} PROPERTIES
      LABELS "host;modules;memory;exact;compaction" TIMEOUT 180 SKIP_RETURN_CODE 77)
  endforeach()
endforeach()

# Implicit textual profiles may have F/DQ without VL/BW: smaller registers
# retain the AVX2 path instead of referencing unavailable VL instructions.
if("AVX512" IN_LIST SIMD_PROFILES)
  add_library(simd_compaction_no_vl_body OBJECT no_vl.cc)
  target_link_libraries(simd_compaction_no_vl_body PRIVATE simd::common)
  simd_target_profile(simd_compaction_no_vl_body AVX512)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    set_property(SOURCE no_vl.cc APPEND PROPERTY COMPILE_OPTIONS /clang:-mno-avx512vl /clang:-mno-avx512bw)
  else()
    set_property(SOURCE no_vl.cc APPEND PROPERTY COMPILE_OPTIONS -mno-avx512vl -mno-avx512bw)
  endif()
  add_executable(simd_compaction_no_vl driver.cc $<TARGET_OBJECTS:simd_compaction_no_vl_body>)
  target_compile_definitions(simd_compaction_no_vl PRIVATE SIMD_TEST_PROFILE=512)
  target_link_libraries(simd_compaction_no_vl PRIVATE simd::common simd::simd)
  set_target_properties(simd_compaction_no_vl simd_compaction_no_vl_body PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  simd_link_clang_cl_asan(simd_compaction_no_vl)
  add_test(NAME simd.compaction.no_vl COMMAND simd_compaction_no_vl)
  set_tests_properties(simd.compaction.no_vl PROPERTIES SKIP_RETURN_CODE 77 LABELS "host;memory;exact;compaction")
endif()
