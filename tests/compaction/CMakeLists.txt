foreach(profile IN ITEMS SCALAR ${NATIVE_PROFILES})
  string(TOLOWER "${profile}" lower)
  if(profile MATCHES "^AVX512")
    set(bits 512)
  elseif(profile STREQUAL "AVX2")
    set(bits 256)
  elseif(profile MATCHES "^NEON")
    set(bits 128)
  else()
    set(bits 0)
  endif()
  foreach(kind IN ITEMS header import)
    set(target native_compaction_${lower}_${kind})
    add_library(${target}_body OBJECT main.cc)
    target_include_directories(${target}_body PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
    add_executable(${target} driver.cc $<TARGET_OBJECTS:${target}_body>)
    target_compile_definitions(${target} PRIVATE NATIVE_TEST_PROFILE=${bits})
    target_link_libraries(${target} PRIVATE native::common native::native)
    set_target_properties(${target} ${target}_body PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
    target_compile_definitions(${target}_body PRIVATE NATIVE_TEST_PROFILE=${bits})
    if(profile STREQUAL "AVX512_BF16" OR profile STREQUAL "NEON_BF16")
      target_compile_definitions(${target} PRIVATE NATIVE_TEST_BF16=1)
      target_compile_definitions(${target}_body PRIVATE NATIVE_TEST_BF16=1)
    endif()
    if(profile STREQUAL "AVX512_FP16")
      target_compile_definitions(${target} PRIVATE NATIVE_TEST_AVX512_FP16=1)
      target_compile_definitions(${target}_body PRIVATE NATIVE_TEST_AVX512_FP16=1)
    endif()
    if(profile STREQUAL "NEON_FP16")
      target_compile_definitions(${target} PRIVATE NATIVE_TEST_FP16=1)
      target_compile_definitions(${target}_body PRIVATE NATIVE_TEST_FP16=1)
    endif()
    target_link_libraries(${target}_body PRIVATE native::common)
    if(kind STREQUAL "import")
      target_compile_definitions(${target}_body PRIVATE NATIVE_TEST_IMPORT=1)
    endif()
    if(NOT profile STREQUAL "SCALAR")
      native_target_profile(${target}_body ${profile})
      target_link_libraries(${target}_body PRIVATE native::${lower})
    endif()
    native_link_clang_cl_asan(${target})
    add_test(NAME native.compaction.${lower}.${kind} COMMAND ${target})
    set_tests_properties(native.compaction.${lower}.${kind} PROPERTIES
      LABELS "host;modules;memory;exact;compaction" TIMEOUT 180 SKIP_RETURN_CODE 77)
  endforeach()
endforeach()

# Explicit feature tags may have F/DQ without VL/BW: smaller registers
# retain the AVX2 path instead of referencing unavailable VL instructions.
if("AVX512" IN_LIST NATIVE_PROFILES)
  add_library(native_compaction_no_vl_body OBJECT no_vl.cc)
  target_link_libraries(native_compaction_no_vl_body PRIVATE native::common)
  native_target_profile(native_compaction_no_vl_body AVX512)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    set_property(SOURCE no_vl.cc APPEND PROPERTY COMPILE_OPTIONS /clang:-mno-avx512vl /clang:-mno-avx512bw)
  else()
    set_property(SOURCE no_vl.cc APPEND PROPERTY COMPILE_OPTIONS -mno-avx512vl -mno-avx512bw)
  endif()
  add_executable(native_compaction_no_vl driver.cc $<TARGET_OBJECTS:native_compaction_no_vl_body>)
  target_compile_definitions(native_compaction_no_vl PRIVATE NATIVE_TEST_PROFILE=512)
  target_link_libraries(native_compaction_no_vl PRIVATE native::common native::native)
  set_target_properties(native_compaction_no_vl native_compaction_no_vl_body PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  native_link_clang_cl_asan(native_compaction_no_vl)
  add_test(NAME native.compaction.no_vl COMMAND native_compaction_no_vl)
  set_tests_properties(native.compaction.no_vl PROPERTIES SKIP_RETURN_CODE 77 LABELS "host;memory;exact;compaction")
endif()
