cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_arm_sm_crypto_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
endif()
foreach(suite IN ITEMS main metadata participation constexpr)
  if(suite STREQUAL "constexpr")
    add_executable(native_test_arm_sm_crypto_${suite} main.cc)
    target_compile_definitions(native_test_arm_sm_crypto_${suite} PRIVATE NATIVE_SM_CONSTANT_TEST=1)
  else()
    add_executable(native_test_arm_sm_crypto_${suite} ${suite}.cc)
  endif()
  target_link_libraries(native_test_arm_sm_crypto_${suite} PRIVATE native::native)
  set_target_properties(native_test_arm_sm_crypto_${suite} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  add_test(NAME native.arm.sm_crypto.${suite} COMMAND native_test_arm_sm_crypto_${suite})
  set_tests_properties(native.arm.sm_crypto.${suite} PROPERTIES SKIP_RETURN_CODE 77 LABELS "arm;sm3;sm4;modules;constexpr")
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  return()
endif()
add_library(native_test_arm_sm_crypto_codegen OBJECT codegen.cc)
target_link_libraries(native_test_arm_sm_crypto_codegen PRIVATE native::native)
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  target_compile_options(native_test_arm_sm_crypto_codegen PRIVATE /O2 /clang:-mcpu=generic /clang:-march=armv8-a)
else()
  target_compile_options(native_test_arm_sm_crypto_codegen PRIVATE -O2 -mcpu=generic -march=armv8-a)
endif()
set_target_properties(native_test_arm_sm_crypto_codegen PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(sm_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(sm_objdump NAMES llvm-objdump HINTS "${sm_compiler_dir}" REQUIRED)
add_test(NAME native.arm.sm_crypto.codegen COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../arm_dotprod/check_equivalence.py"
  --objdump "${sm_objdump}" --output "${CMAKE_CURRENT_BINARY_DIR}/paired-$<CONFIG>.json"
  $<TARGET_OBJECTS:native_test_arm_sm_crypto_codegen> COMMAND_EXPAND_LISTS)
set_tests_properties(native.arm.sm_crypto.codegen PROPERTIES LABELS "arm;sm3;sm4;codegen")

# Test every weak overload and every runtime instruction boundary separately.
foreach(op RANGE 0 8)
  foreach(kind RANGE 0 4)
    if(kind GREATER 2 AND (op LESS 1 OR op GREATER 4))
      continue()
    endif()
    set(target native_test_arm_sm_crypto_reject_${op}_${kind})
    add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
    target_link_libraries(${target} PRIVATE native::native)
    target_compile_definitions(${target} PRIVATE NATIVE_SM_REJECT_OP=${op} NATIVE_SM_REJECT_KIND=${kind})
    set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON)
    if(kind EQUAL 0)
      set(diagnostic "call to consteval function.*not a constant expression")
    elseif(kind EQUAL 1)
      set(diagnostic "requires target feature.*sm4|target specific option mismatch")
    else()
      set(diagnostic "no matching function")
    endif()
    add_test(NAME native.arm.sm_crypto.reject_${op}_${kind} COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../arm_fp16fml/check_negative.py"
      --cmake "${CMAKE_COMMAND}" --build "${CMAKE_BINARY_DIR}" --target "${target}"
      --config "$<CONFIG>" --diagnostic "${diagnostic}"
      --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${op}-${kind}-$<CONFIG>.log")
    set_tests_properties(native.arm.sm_crypto.reject_${op}_${kind} PROPERTIES
      LABELS "arm;sm3;sm4;negative" RUN_SERIAL TRUE)
  endforeach()
endforeach()

foreach(enabled RANGE 0 1)
  set(target native_test_arm_sm_crypto_minimum_${enabled})
  add_executable(${target} minimum.cc)
  target_link_libraries(${target} PRIVATE native::minimal)
  target_compile_definitions(${target} PRIVATE NATIVE_SM_EXPECT_ENABLED=${enabled})
  if(enabled)
    set(march armv8-a+sm4)
  else()
    set(march armv8-a)
  endif()
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /clang:-mcpu=generic /clang:-march=${march})
  else()
    target_compile_options(${target} PRIVATE -mcpu=generic -march=${march})
  endif()
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  # No optional instructions execute: this TU contains only compile-time metadata checks.
  add_test(NAME native.arm.sm_crypto.minimum_${enabled} COMMAND ${target})
  set_tests_properties(native.arm.sm_crypto.minimum_${enabled} PROPERTIES LABELS "arm;sm3;sm4;metadata")
endforeach()
