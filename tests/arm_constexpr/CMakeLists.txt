if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_arm_constexpr_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  return()
endif()
function(arm_constant_baseline target)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /O2 /clang:-mcpu=generic /clang:-march=armv8-a+nocrc+nosha2)
  else()
    target_compile_options(${target} PRIVATE -O2 -mcpu=generic -march=armv8-a+nocrc+nosha2)
  endif()
  set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
endfunction()
foreach(interface IN ITEMS module hub)
  set(target native_test_arm_constexpr_${interface})
  add_executable(${target} module.cc)
  target_link_libraries(${target} PRIVATE native::native)
  if(interface STREQUAL "hub")
    target_compile_definitions(${target} PRIVATE NATIVE_ARM_CONSTEXPR_HUB=1)
  endif()
  arm_constant_baseline(${target})
  if(COMMAND native_link_clang_cl_asan)
    native_link_clang_cl_asan(${target})
  endif()
  add_test(NAME native.arm.constexpr.${interface} COMMAND ${target})
  set_tests_properties(native.arm.constexpr.${interface} PROPERTIES LABELS "host;arm;modules;constexpr")
endforeach()

# The ordinary CRC header retains its C++20 and explicit-ISA contract.
add_executable(native_test_arm_constexpr_header header.cc)
target_link_libraries(native_test_arm_constexpr_header PRIVATE native::headers)
set_target_properties(native_test_arm_constexpr_header PROPERTIES
  CXX_STANDARD 20 CXX_STANDARD_REQUIRED TRUE CXX_SCAN_FOR_MODULES OFF)
arm_constant_baseline(native_test_arm_constexpr_header)
if(COMMAND native_link_clang_cl_asan)
  native_link_clang_cl_asan(native_test_arm_constexpr_header)
endif()
add_test(NAME native.arm.constexpr.header COMMAND native_test_arm_constexpr_header)
set_tests_properties(native.arm.constexpr.header PROPERTIES LABELS "host;arm;headers;constexpr")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(kind RANGE 0 10)
  set(target native_arm_constexpr_reject_${kind})
  add_library(${target} OBJECT EXCLUDE_FROM_ALL reject.cc)
  target_link_libraries(${target} PRIVATE native::native)
  target_compile_definitions(${target} PRIVATE NATIVE_ARM_CONSTEXPR_REJECT=${kind})
  arm_constant_baseline(${target})
  if(kind LESS 9)
    set(diagnostic "call to consteval function.*is not a constant expression")
  else()
    set(diagnostic "requires target feature|target specific option mismatch")
  endif()
  add_test(NAME native.arm.constexpr.reject_${kind} COMMAND "${Python3_EXECUTABLE}"
    "${PROJECT_SOURCE_DIR}/tests/source_targets/check_negative.py" --cmake "${CMAKE_COMMAND}"
    --build "${CMAKE_BINARY_DIR}" --target "${target}" --config "$<CONFIG>"
    --diagnostic "${diagnostic}" --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${kind}-$<CONFIG>.log")
  set_tests_properties(native.arm.constexpr.reject_${kind} PROPERTIES
    RUN_SERIAL TRUE TIMEOUT 180 LABELS "host;arm;constexpr;negative")
endforeach()
