cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(simd_kernel_policy_refinement LANGUAGES CXX)
  find_package(simd CONFIG REQUIRED COMPONENTS simd headers)
  enable_testing()
endif()
# No per-fixture ISA flags: native entry points carry literal target attributes.
set(kernel_kinds module)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/simd/vec.h")
  list(PREPEND kernel_kinds header)
endif()
foreach(kind IN LISTS kernel_kinds)
  add_executable(simd_test_kernel_refinement_${kind} native.cc)
  target_link_libraries(simd_test_kernel_refinement_${kind} PRIVATE simd::simd)
  if(kind STREQUAL "module")
    target_compile_definitions(simd_test_kernel_refinement_${kind} PRIVATE SIMD_TEST_IMPORT=1)
  else()
    target_include_directories(simd_test_kernel_refinement_${kind} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
  endif()
  if(COMMAND simd_link_clang_cl_asan)
    simd_link_clang_cl_asan(simd_test_kernel_refinement_${kind})
  endif()
  add_test(NAME simd.kernel_refinement.${kind} COMMAND simd_test_kernel_refinement_${kind})
  set_tests_properties(simd.kernel_refinement.${kind} PROPERTIES
    SKIP_RETURN_CODE 77 LABELS "host;kernel_refinement" TIMEOUT 120)
endforeach()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$" AND NOT SIMD_ENABLE_ASAN)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  get_filename_component(kernel_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
  find_program(kernel_objdump NAMES llvm-objdump HINTS "${kernel_compiler_dir}" REQUIRED)
  add_library(simd_kernel_refinement_codegen OBJECT codegen.cc)
  target_link_libraries(simd_kernel_refinement_codegen PRIVATE simd::simd)
  set_target_properties(simd_kernel_refinement_codegen PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  add_test(NAME simd.kernel_refinement.codegen COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/check_codegen.py" --objdump "${kernel_objdump}"
    --output "${CMAKE_CURRENT_BINARY_DIR}/kernel-$<CONFIG>.disassembly"
    $<TARGET_OBJECTS:simd_kernel_refinement_codegen> COMMAND_EXPAND_LISTS)
  set_tests_properties(simd.kernel_refinement.codegen PROPERTIES LABELS "host;kernel_refinement" TIMEOUT 120)
endif()
