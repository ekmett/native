if(NOT "AVX2" IN_LIST SIMD_PROFILES OR NOT "AVX512" IN_LIST SIMD_PROFILES)
  return()
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
  return()
endif()
foreach(profile IN ITEMS AVX2 AVX512)
  string(TOLOWER "${profile}" lower)
  if(profile STREQUAL "AVX2")
    set(bits 256)
  else()
    set(bits 512)
  endif()
  foreach(kind IN ITEMS header import)
    set(target simd_profile_${lower}_${kind})
    add_library(${target} OBJECT bridge.cc)
    target_link_libraries(${target} PRIVATE simd::common simd::${lower})
    simd_target_profile(${target} ${profile})
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
    if(kind STREQUAL "header")
      set(interface 0)
    else()
      set(interface 1)
    endif()
    target_compile_definitions(${target} PRIVATE SIMD_TEST_PROFILE=${bits}
      SIMD_TEST_INTERFACE=${interface} SIMD_TEST_ENTRY=profile_${lower}_${kind})
    list(APPEND profile_objects $<TARGET_OBJECTS:${target}>)
  endforeach()
endforeach()
add_library(simd_profile_identity OBJECT identity.cc)
target_link_libraries(simd_profile_identity PRIVATE simd::common simd::avx2 simd::avx512)
simd_target_profile(simd_profile_identity AVX512)
add_executable(simd_test_profiles main.cc ${profile_objects} $<TARGET_OBJECTS:simd_profile_identity>)
target_link_libraries(simd_test_profiles PRIVATE simd::simd simd::common)
simd_link_clang_cl_asan(simd_test_profiles)
set_target_properties(simd_test_profiles PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
foreach(mode IN ITEMS all avx2 avx512 none)
  add_test(NAME simd.profiles.${mode} COMMAND simd_test_profiles ${mode})
  set_tests_properties(simd.profiles.${mode} PROPERTIES SKIP_RETURN_CODE 77
    LABELS "host;modules;profiles;exact" TIMEOUT 180)
endforeach()

target_include_directories(simd_test_profiles PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
