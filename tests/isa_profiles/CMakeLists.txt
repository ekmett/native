foreach(profile IN ITEMS AVX2 AVX512)
  string(TOLOWER "${profile}" lower)
  if(profile STREQUAL "AVX2")
    set(bits 256)
  else()
    set(bits 512)
  endif()
  foreach(kind IN ITEMS header import)
    set(target native_profile_${lower}_${kind})
    add_library(${target} OBJECT bridge.cc)
    target_link_libraries(${target} PRIVATE native::common native::${lower})
    native_target_profile(${target} ${profile})
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
    if(kind STREQUAL "header")
      set(interface 0)
    else()
      set(interface 1)
    endif()
    target_compile_definitions(${target} PRIVATE NATIVE_TEST_PROFILE=${bits}
      NATIVE_TEST_INTERFACE=${interface} NATIVE_TEST_ENTRY=profile_${lower}_${kind})
    list(APPEND profile_objects $<TARGET_OBJECTS:${target}>)
  endforeach()
endforeach()
add_library(native_profile_identity OBJECT identity.cc)
target_link_libraries(native_profile_identity PRIVATE native::common native::avx2 native::avx512)
native_target_profile(native_profile_identity AVX512)
add_executable(native_test_profiles main.cc ${profile_objects} $<TARGET_OBJECTS:native_profile_identity>)
target_link_libraries(native_test_profiles PRIVATE native::native native::common)
native_link_clang_cl_asan(native_test_profiles)
set_target_properties(native_test_profiles PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
foreach(mode IN ITEMS all avx2 avx512 none)
  add_test(NAME native.profiles.${mode} COMMAND native_test_profiles ${mode})
  set_tests_properties(native.profiles.${mode} PROPERTIES SKIP_RETURN_CODE 77
    LABELS "host;modules;profiles;exact" TIMEOUT 180)
endforeach()

target_include_directories(native_test_profiles PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
