cmake_minimum_required(VERSION 4.4)
project(native_omnibus_consumer LANGUAGES CXX)
find_package(native CONFIG REQUIRED COMPONENTS native)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(OMNIBUS_PCH "Use a consumer-owned standard-header PCH" ON)
option(OMNIBUS_IPO "Use IPO on native kernels" ON)
enable_testing()
set(OMNIBUS_CONSUMER_PROFILE "" CACHE STRING
  "Optional single-profile override for compiler/BMI diagnostics; empty uses each kernel profile")
add_executable(omnibus_archive_only archive_only.cc)
target_link_libraries(omnibus_archive_only PRIVATE native::native)
set_target_properties(omnibus_archive_only PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_test(NAME omnibus.archive_only COMMAND omnibus_archive_only)
add_executable(omnibus_baseline baseline.cc)
target_link_libraries(omnibus_baseline PRIVATE native::native)
set_target_properties(omnibus_baseline PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_test(NAME omnibus.baseline COMMAND omnibus_baseline)
foreach(profile AVX2 AVX512 NEON NEON_FP16 NEON_BF16)
  string(TOLOWER "${profile}" lower)
  if(NOT TARGET native::${lower})
    continue()
  endif()
  add_library(kernel_${lower} STATIC kernel.cc granular.cc)
  target_link_libraries(kernel_${lower} PRIVATE native::native)
  if(OMNIBUS_CONSUMER_PROFILE)
    native_target_profile(kernel_${lower} ${OMNIBUS_CONSUMER_PROFILE})
  else()
    native_target_profile(kernel_${lower} ${profile})
  endif()
  target_compile_definitions(kernel_${lower} PRIVATE TEST_${profile}=1)
  if(TARGET native::avx2 AND TARGET native::avx512)
    target_compile_definitions(kernel_${lower} PRIVATE TEST_BOTH_X86=1)
  endif()
  set_target_properties(kernel_${lower} PROPERTIES INTERPROCEDURAL_OPTIMIZATION ${OMNIBUS_IPO})
  if(OMNIBUS_PCH)
    target_precompile_headers(kernel_${lower} PRIVATE <array> <concepts> <cstddef> <type_traits> <utility>)
  endif()
  add_executable(omnibus_${lower} main.cc)
  target_compile_definitions(omnibus_${lower} PRIVATE TEST_${profile}=1)
  target_compile_definitions(omnibus_${lower} PRIVATE TEST_REQUIRED_${profile}=1)
  target_link_libraries(omnibus_${lower} PRIVATE kernel_${lower} native::native)
  set_target_properties(omnibus_${lower} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  add_test(NAME omnibus.${lower} COMMAND omnibus_${lower})
  set_tests_properties(omnibus.${lower} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
