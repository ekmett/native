cmake_minimum_required(VERSION 4.4)
project(simd_cpuid_module_check LANGUAGES CXX)
string(REPLACE " -TP " " " CMAKE_CXX_COMPILE_OBJECT "${CMAKE_CXX_COMPILE_OBJECT}")
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SIMD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SIMD_ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(SIMD_ENABLE_IPO OFF CACHE BOOL "" FORCE)
get_filename_component(simd_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${simd_root}" simd EXCLUDE_FROM_ALL)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
  message(FATAL_ERROR "The CPUID/wait fixture requires x86-64")
endif()
enable_testing()
add_executable(cpuid_import import.cc)
add_executable(cpuid_admission admission.cc)
add_executable(wait_noexcept "${simd_root}/tests/core_regression/noexcept_wait.cc")
add_library(wait_codegen OBJECT codegen.cc)
foreach(target IN ITEMS cpuid_import cpuid_admission wait_noexcept wait_codegen)
  target_link_libraries(${target} PRIVATE simd::common)
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
simd_link_clang_cl_asan(cpuid_import)
simd_link_clang_cl_asan(cpuid_admission)
simd_link_clang_cl_asan(wait_noexcept)
add_test(NAME cpuid.import COMMAND cpuid_import)
add_test(NAME cpuid.admission COMMAND cpuid_admission)
add_test(NAME wait.noexcept COMMAND wait_noexcept)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/wait-codegen-object.txt" CONTENT "$<TARGET_OBJECTS:wait_codegen>\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/common-archive.txt" CONTENT "$<TARGET_FILE:simd_minimal>\n")
