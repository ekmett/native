cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_x86_wait_check LANGUAGES CXX)
  set(NATIVE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../.. native EXCLUDE_FROM_ALL)
  enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(native_test_x86_wait main.cc)
target_link_libraries(native_test_x86_wait PRIVATE native::minimal)
add_test(NAME native.x86.wait.runtime COMMAND native_test_x86_wait)
set_tests_properties(native.x86.wait.runtime PROPERTIES SKIP_RETURN_CODE 77)
foreach(kind RANGE 0 3)
  set(target native_test_x86_wait_negative_${kind})
  if(kind EQUAL 0)
    add_library(${target} OBJECT negative.cc)
  else()
    add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
  endif()
  target_link_libraries(${target} PRIVATE native::minimal)
  target_compile_definitions(${target} PRIVATE NATIVE_WAIT_NEGATIVE=${kind})
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /clang:-O2)
  else()
    target_compile_options(${target} PRIVATE -O2)
  endif()
  if(kind GREATER 0)
    set(diagnostic "constraints not satisfied")
    if(kind EQUAL 2)
      set(diagnostic "requires target feature")
    endif()
    add_test(NAME native.x86.wait.reject_${kind} COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../source_targets/check_negative.py"
      --cmake "${CMAKE_COMMAND}" --build "${CMAKE_BINARY_DIR}" --target "${target}"
      --config "$<CONFIG>" --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${kind}.log"
      --diagnostic "${diagnostic}")
    set_tests_properties(native.x86.wait.reject_${kind} PROPERTIES RUN_SERIAL TRUE)
  endif()
endforeach()
get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(wait_objdump NAMES llvm-objdump HINTS "${compiler_directory}" REQUIRED)
foreach(mode IN ITEMS raw public)
  add_library(native_test_x86_wait_codegen_${mode} OBJECT codegen.cc)
  target_link_libraries(native_test_x86_wait_codegen_${mode} PRIVATE native::minimal)
  if(mode STREQUAL "raw")
    target_compile_definitions(native_test_x86_wait_codegen_${mode} PRIVATE NATIVE_WAIT_RAW=1)
  endif()
  set_target_properties(native_test_x86_wait_codegen_${mode} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
endforeach()
add_test(NAME native.x86.wait.codegen COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../x86_codegen_pairs.py" --objdump "${wait_objdump}"
  --raw "$<TARGET_OBJECTS:native_test_x86_wait_codegen_raw>"
  --public "$<TARGET_OBJECTS:native_test_x86_wait_codegen_public>" --expected 1
  --output "${CMAKE_CURRENT_BINARY_DIR}/paired-codegen.txt")
