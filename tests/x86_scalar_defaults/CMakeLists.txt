function(scalar_defaults_options target)
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON
    INTERPROCEDURAL_OPTIMIZATION FALSE)
  foreach(option IN LISTS ARGN)
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
      target_compile_options(${target} PRIVATE "/clang:${option}")
    else()
      target_compile_options(${target} PRIVATE "${option}")
    endif()
  endforeach()
endfunction()

set(weak_options -march=x86-64 -mno-bmi -mno-bmi2 -mno-popcnt -mno-lzcnt -mno-crc32 -mno-f16c)
set(strong_options -march=x86-64 -mbmi -mbmi2 -mpopcnt -mlzcnt -mcrc32 -mf16c)
set(sse42_no_popcnt_options -march=x86-64 -msse4.2 -mno-popcnt -mno-bmi -mno-bmi2 -mno-lzcnt -mno-f16c)
add_library(native_scalar_defaults_header OBJECT header.cc)

target_compile_features(native_scalar_defaults_header PRIVATE cxx_std_26)
scalar_defaults_options(native_scalar_defaults_header ${strong_options})
set_target_properties(native_scalar_defaults_header PROPERTIES CXX_SCAN_FOR_MODULES OFF)

foreach(mode IN ITEMS weak strong sse42_no_popcnt)
  # Compile the real owning module contents with independent provider options.
  # Only their names change, avoiding duplicate public providers in one graph.
  set(providers)
  set(imports "")
  foreach(family IN ITEMS bmi1 bmi2 popcnt lzcnt crc32c f16c)
    set(source "${PROJECT_SOURCE_DIR}/src/native.x86.${family}.ccm")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(READ "${source}" contents)
    set(name "native.test.scalar_defaults.${mode}.${family}")
    string(REPLACE "export module native.x86.${family};" "export module ${name};" contents "${contents}")
    set(provider "${CMAKE_CURRENT_BINARY_DIR}/${mode}_${family}.ccm")
    file(WRITE "${provider}" "${contents}")
    list(APPEND providers "${provider}")
    string(APPEND imports "import ${name};\n")
  endforeach()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${mode}_imports.h" "${imports}")
  add_library(native_scalar_defaults_${mode}_provider STATIC)
  target_sources(native_scalar_defaults_${mode}_provider PUBLIC FILE_SET CXX_MODULES
    BASE_DIRS "${CMAKE_CURRENT_BINARY_DIR}" FILES ${providers})
  target_link_libraries(native_scalar_defaults_${mode}_provider PUBLIC native::native)
  scalar_defaults_options(native_scalar_defaults_${mode}_provider ${${mode}_options})
  configure_file(consumer.cc.in "${mode}_consumer.cc" @ONLY)
  add_library(native_scalar_defaults_${mode} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${mode}_consumer.cc")
  target_link_libraries(native_scalar_defaults_${mode} PRIVATE native_scalar_defaults_${mode}_provider)
  target_include_directories(native_scalar_defaults_${mode} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  scalar_defaults_options(native_scalar_defaults_${mode} ${${mode}_options})
endforeach()

set(negative_kinds feature_bmi1 feature_bmi2 feature_popcnt feature_lzcnt feature_crc32 feature_f16c
  target_bmi1 target_bmi2 target_popcnt target_lzcnt target_crc32 target_f16c
  rorx_range rorx_runtime half_range half_runtime)
set(index 0)
foreach(kind IN LISTS negative_kinds)
  set(target native_scalar_defaults_reject_${kind})
  add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(${target} PRIVATE NATIVE_SCALAR_NEGATIVE=${index})
  if(kind MATCHES "^feature")
    target_link_libraries(${target} PRIVATE native_scalar_defaults_weak_provider)
    scalar_defaults_options(${target} ${weak_options})
  else()
    target_link_libraries(${target} PRIVATE native_scalar_defaults_strong_provider)
    scalar_defaults_options(${target} ${strong_options})
  endif()
  if(kind MATCHES "^target")
    set(diagnostic "requires target feature|target specific option mismatch")
  elseif(kind MATCHES "runtime$")
    set(diagnostic "invalid explicitly-specified argument|non-type template argument is not a constant expression")
  else()
    set(diagnostic "constraints not satisfied")
  endif()
  add_test(NAME native.x86.scalar_defaults.reject_${kind} COMMAND "${Python3_EXECUTABLE}"
    "${PROJECT_SOURCE_DIR}/tests/source_targets/check_negative.py" --cmake "${CMAKE_COMMAND}"
    --build "${CMAKE_BINARY_DIR}" --target "${target}" --config "$<CONFIG>"
    --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${kind}-$<CONFIG>.log" --diagnostic "${diagnostic}")
  set_tests_properties(native.x86.scalar_defaults.reject_${kind} PROPERTIES RUN_SERIAL TRUE
    LABELS "host;negative;modules;baseline;x86")
  math(EXPR index "${index}+1")
endforeach()
