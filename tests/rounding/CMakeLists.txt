cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(simd_rounding_check LANGUAGES CXX)
  find_package(simd CONFIG REQUIRED)
  enable_testing()
  set(SIMD_ROUNDING_PROFILES "AVX2;AVX512" CACHE STRING "Profiles to exercise")
else()
  set(SIMD_ROUNDING_PROFILES "${SIMD_PROFILES}")
endif()
foreach(profile IN ITEMS SCALAR ${SIMD_ROUNDING_PROFILES})
  string(TOLOWER "${profile}" lower)
  if(profile STREQUAL "SCALAR")
    set(bits 0)
  elseif(profile STREQUAL "AVX2")
    set(bits 256)
  elseif(profile STREQUAL "AVX512")
    set(bits 512)
  elseif(profile STREQUAL "NEON")
    set(bits 128)
  else()
    message(FATAL_ERROR "Unknown rounding test profile ${profile}")
  endif()
  foreach(kind IN ITEMS header import)
    set(target simd_rounding_${lower}_${kind})
    add_executable(${target} rounding.cc)
    target_link_libraries(${target} PRIVATE simd::simd simd::common)
    if(NOT profile STREQUAL "SCALAR")
      target_link_libraries(${target} PRIVATE simd::${lower})
      simd_target_profile(${target} ${profile})
    endif()
    target_compile_definitions(${target} PRIVATE SIMD_ROUNDING_PROFILE=${bits})
    if(kind STREQUAL "header")
      target_compile_definitions(${target} PRIVATE SIMD_ROUNDING_HEADER=1)
    endif()
    target_compile_features(${target} PRIVATE cxx_std_26)
    set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
      target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off -frounding-math)
    endif()
    if(COMMAND simd_link_clang_cl_asan)
      simd_link_clang_cl_asan(${target})
    endif()
    add_test(NAME simd.rounding.${lower}.${kind} COMMAND ${target})
    set_tests_properties(simd.rounding.${lower}.${kind} PROPERTIES LABELS "host;modules;rounding;exact")
  endforeach()
endforeach()
add_executable(simd_rounding_adl adl.cc)
target_link_libraries(simd_rounding_adl PRIVATE simd::simd simd::common)
set_target_properties(simd_rounding_adl PROPERTIES CXX_SCAN_FOR_MODULES ON)
if(COMMAND simd_link_clang_cl_asan)
  simd_link_clang_cl_asan(simd_rounding_adl)
endif()
add_test(NAME simd.rounding.adl COMMAND simd_rounding_adl)
