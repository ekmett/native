cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_rounding_check LANGUAGES CXX)
  find_package(native CONFIG REQUIRED)
  enable_testing()
endif()
# Building an ISA provider does not admit that ISA for execution. Match the
# configured test profile unless the runner explicitly selects several profiles.
if(DEFINED NATIVE_TEST_ISA)
  set(native_rounding_default_profile "${NATIVE_TEST_ISA}")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(native_rounding_default_profile NEON)
else()
  set(native_rounding_default_profile AVX2)
endif()
set(NATIVE_ROUNDING_PROFILES "${native_rounding_default_profile}" CACHE STRING "Admitted rounding test profiles")
foreach(profile IN ITEMS SCALAR ${NATIVE_ROUNDING_PROFILES})
  string(TOLOWER "${profile}" lower)
  if(profile STREQUAL "SCALAR")
    set(bits 0)
  elseif(profile STREQUAL "AVX2")
    set(bits 256)
  elseif(profile STREQUAL "AVX512")
    set(bits 512)
  elseif(profile STREQUAL "NEON")
    set(bits 128)
  else()
    message(FATAL_ERROR "Unknown rounding test profile ${profile}")
  endif()
  foreach(kind IN ITEMS header import)
    set(target native_rounding_${lower}_${kind})
    add_executable(${target} rounding.cc)
    target_link_libraries(${target} PRIVATE native::native native::common)
    if(NOT profile STREQUAL "SCALAR")
      target_link_libraries(${target} PRIVATE native::${lower})
      native_target_profile(${target} ${profile})
    endif()
    target_compile_definitions(${target} PRIVATE NATIVE_ROUNDING_PROFILE=${bits})
    if(kind STREQUAL "header")
      target_compile_definitions(${target} PRIVATE NATIVE_ROUNDING_HEADER=1)
    endif()
    target_compile_features(${target} PRIVATE cxx_std_26)
    set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
      target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off -frounding-math)
    endif()
    if(COMMAND native_link_clang_cl_asan)
      native_link_clang_cl_asan(${target})
    endif()
    add_test(NAME native.rounding.${lower}.${kind} COMMAND ${target})
    set_tests_properties(native.rounding.${lower}.${kind} PROPERTIES LABELS "host;modules;rounding;exact")
  endforeach()
endforeach()
add_executable(native_rounding_adl adl.cc)
target_link_libraries(native_rounding_adl PRIVATE native::native native::common)
set_target_properties(native_rounding_adl PROPERTIES CXX_SCAN_FOR_MODULES ON)
if(COMMAND native_link_clang_cl_asan)
  native_link_clang_cl_asan(native_rounding_adl)
endif()
add_test(NAME native.rounding.adl COMMAND native_rounding_adl)
