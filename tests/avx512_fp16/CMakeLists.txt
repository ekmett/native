cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_avx512_fp16_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
  set(avx512_fp16_standalone TRUE)
endif()
if(NOT TARGET native::avx512_fp16)
  if(avx512_fp16_standalone)
    message(FATAL_ERROR "This fixture requires a package for a host supporting AVX512_FP16.")
  endif()
  return()
endif()
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT "AVX512_FP16" IN_LIST NATIVE_PROFILES)
  return()
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(native_avx512_fp16_kernel OBJECT kernel.cc)
target_link_libraries(native_avx512_fp16_kernel PRIVATE native::minimal native::avx512_fp16)
native_target_profile(native_avx512_fp16_kernel AVX512_FP16)
target_include_directories(native_avx512_fp16_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
set_target_properties(native_avx512_fp16_kernel PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_library(native_avx512_fp16_omnibus OBJECT omnibus.cc)
target_link_libraries(native_avx512_fp16_omnibus PRIVATE native::native native::minimal)
native_target_omnibus(native_avx512_fp16_omnibus)
target_precompile_headers(native_avx512_fp16_omnibus PRIVATE <array> <concepts> <cstddef> <type_traits>)
add_executable(native_test_avx512_fp16 main.cc $<TARGET_OBJECTS:native_avx512_fp16_kernel>)
target_link_libraries(native_test_avx512_fp16 PRIVATE native::minimal native::avx512_fp16)
set_target_properties(native_test_avx512_fp16 PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_dependencies(native_test_avx512_fp16 native_avx512_fp16_omnibus)
if(COMMAND native_link_clang_cl_asan)
  native_link_clang_cl_asan(native_test_avx512_fp16)
endif()
if(TARGET native::avx512)
  add_library(native_avx512_fp16_identity OBJECT identity.cc)
  target_link_libraries(native_avx512_fp16_identity PRIVATE native::minimal native::avx512 native::avx512_fp16)
  native_target_profile(native_avx512_fp16_identity AVX512_FP16)
  add_dependencies(native_test_avx512_fp16 native_avx512_fp16_identity)
endif()
add_executable(native_avx512_fp16_admission ../cpuid_module/admission.cc)
target_link_libraries(native_avx512_fp16_admission PRIVATE native::minimal)
add_dependencies(native_test_avx512_fp16 native_avx512_fp16_admission)
add_test(NAME native.avx512_fp16.admission COMMAND native_avx512_fp16_admission)
set_tests_properties(native.avx512_fp16.admission PROPERTIES LABELS "host;modules;avx512_fp16;baseline")
if(COMMAND native_link_clang_cl_asan)
  native_link_clang_cl_asan(native_avx512_fp16_admission)
endif()
add_test(NAME native.avx512_fp16.feature_boundaries COMMAND "${CMAKE_COMMAND}"
  "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DNINJA=${CMAKE_MAKE_PROGRAM}" -P "${CMAKE_CURRENT_SOURCE_DIR}/feature_boundaries.cmake")
set_tests_properties(native.avx512_fp16.feature_boundaries PROPERTIES LABELS "host;modules;avx512_fp16;profiles")
foreach(mode IN ITEMS none native)
  add_test(NAME native.avx512_fp16.${mode} COMMAND native_test_avx512_fp16 ${mode})
  set_tests_properties(native.avx512_fp16.${mode} PROPERTIES SKIP_RETURN_CODE 77
    LABELS "host;modules;avx512_fp16;exact" TIMEOUT 180)
endforeach()
get_filename_component(avx512_fp16_compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(NATIVE_AVX512_FP16_OBJDUMP NAMES llvm-objdump HINTS "${avx512_fp16_compiler_directory}")
get_target_property(avx512_fp16_usage_options native::minimal INTERFACE_COMPILE_OPTIONS)
# Sanitizer instrumentation deliberately introduces calls; inspect ordinary objects.
if(NATIVE_AVX512_FP16_OBJDUMP AND NOT avx512_fp16_usage_options MATCHES "sanitize=address")
  add_test(NAME native.avx512_fp16.codegen COMMAND "${CMAKE_COMMAND}"
    "-DOBJDUMP=${NATIVE_AVX512_FP16_OBJDUMP}" "-DOBJECT=$<TARGET_OBJECTS:native_avx512_fp16_kernel>"
    "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/codegen.txt"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/codegen.cmake")
  set_tests_properties(native.avx512_fp16.codegen PROPERTIES LABELS "host;modules;avx512_fp16;codegen")
endif()
if(TARGET native::avx512_bf16)
  target_compile_definitions(native_avx512_fp16_omnibus PRIVATE TEST_OMNIBUS_BF16=1)
endif()
if(avx512_fp16_standalone AND CMAKE_GENERATOR STREQUAL "Ninja")
  set(fp16_bmi_modules native.scalar native.wide native.simd native.numerics native.x86.features native.features native)
  if(TARGET native::avx512_bf16)
    list(APPEND fp16_bmi_modules native)
  endif()
  add_test(NAME native.avx512_fp16.shared_bmi COMMAND "${CMAKE_COMMAND}"
    "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DMODULES=${fp16_bmi_modules}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/../profile_package/check_bmi.cmake")
  set_tests_properties(native.avx512_fp16.shared_bmi PROPERTIES LABELS "host;modules;avx512_fp16;profiles")
endif()
