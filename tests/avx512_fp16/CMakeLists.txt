cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(simd_avx512_fp16_consumer LANGUAGES CXX)
  find_package(simd CONFIG REQUIRED COMPONENTS simd)
  enable_testing()
  set(avx512_fp16_standalone TRUE)
endif()
if(NOT TARGET simd::avx512_fp16)
  if(avx512_fp16_standalone)
    message(FATAL_ERROR "This fixture requires a package configured with AVX512_FP16.")
  endif()
  return()
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(simd_avx512_fp16_kernel OBJECT kernel.cc)
target_link_libraries(simd_avx512_fp16_kernel PRIVATE simd::minimal simd::avx512_fp16)
simd_target_profile(simd_avx512_fp16_kernel AVX512_FP16)
target_include_directories(simd_avx512_fp16_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
set_target_properties(simd_avx512_fp16_kernel PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_library(simd_avx512_fp16_omnibus OBJECT omnibus.cc)
target_link_libraries(simd_avx512_fp16_omnibus PRIVATE simd::simd simd::minimal)
simd_target_omnibus(simd_avx512_fp16_omnibus)
target_precompile_headers(simd_avx512_fp16_omnibus PRIVATE <array> <concepts> <cstddef> <type_traits>)
add_executable(simd_test_avx512_fp16 main.cc $<TARGET_OBJECTS:simd_avx512_fp16_kernel>)
target_link_libraries(simd_test_avx512_fp16 PRIVATE simd::minimal simd::avx512_fp16)
set_target_properties(simd_test_avx512_fp16 PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_dependencies(simd_test_avx512_fp16 simd_avx512_fp16_omnibus)
if(COMMAND simd_link_clang_cl_asan)
  simd_link_clang_cl_asan(simd_test_avx512_fp16)
endif()
if(TARGET simd::avx512)
  add_library(simd_avx512_fp16_identity OBJECT identity.cc)
  target_link_libraries(simd_avx512_fp16_identity PRIVATE simd::minimal simd::avx512 simd::avx512_fp16)
  simd_target_profile(simd_avx512_fp16_identity AVX512_FP16)
  add_dependencies(simd_test_avx512_fp16 simd_avx512_fp16_identity)
endif()
add_executable(simd_avx512_fp16_admission ../cpuid_module/admission.cc)
target_link_libraries(simd_avx512_fp16_admission PRIVATE simd::minimal)
add_dependencies(simd_test_avx512_fp16 simd_avx512_fp16_admission)
add_test(NAME simd.avx512_fp16.admission COMMAND simd_avx512_fp16_admission)
set_tests_properties(simd.avx512_fp16.admission PROPERTIES LABELS "host;modules;avx512_fp16;baseline")
if(COMMAND simd_link_clang_cl_asan)
  simd_link_clang_cl_asan(simd_avx512_fp16_admission)
endif()
add_test(NAME simd.avx512_fp16.feature_boundaries COMMAND "${CMAKE_COMMAND}"
  "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DNINJA=${CMAKE_MAKE_PROGRAM}" -P "${CMAKE_CURRENT_SOURCE_DIR}/feature_boundaries.cmake")
set_tests_properties(simd.avx512_fp16.feature_boundaries PROPERTIES LABELS "host;modules;avx512_fp16;profiles")
foreach(mode IN ITEMS none native)
  add_test(NAME simd.avx512_fp16.${mode} COMMAND simd_test_avx512_fp16 ${mode})
  set_tests_properties(simd.avx512_fp16.${mode} PROPERTIES SKIP_RETURN_CODE 77
    LABELS "host;modules;avx512_fp16;exact" TIMEOUT 180)
endforeach()
get_filename_component(avx512_fp16_compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(SIMD_AVX512_FP16_OBJDUMP NAMES llvm-objdump HINTS "${avx512_fp16_compiler_directory}")
get_target_property(avx512_fp16_usage_options simd::minimal INTERFACE_COMPILE_OPTIONS)
# Sanitizer instrumentation deliberately introduces calls; inspect ordinary objects.
if(SIMD_AVX512_FP16_OBJDUMP AND NOT avx512_fp16_usage_options MATCHES "sanitize=address")
  add_test(NAME simd.avx512_fp16.codegen COMMAND "${CMAKE_COMMAND}"
    "-DOBJDUMP=${SIMD_AVX512_FP16_OBJDUMP}" "-DOBJECT=$<TARGET_OBJECTS:simd_avx512_fp16_kernel>"
    "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/codegen.txt"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/codegen.cmake")
  set_tests_properties(simd.avx512_fp16.codegen PROPERTIES LABELS "host;modules;avx512_fp16;codegen")
endif()
if(TARGET simd::avx512_bf16)
  target_compile_definitions(simd_avx512_fp16_omnibus PRIVATE TEST_OMNIBUS_BF16=1)
endif()
if(avx512_fp16_standalone AND CMAKE_GENERATOR STREQUAL "Ninja")
  set(fp16_bmi_modules simd.scalar simd.wide simd.numerics simd.cpuid simd.avx512_fp16)
  if(TARGET simd::avx512_bf16)
    list(APPEND fp16_bmi_modules simd.avx512_bf16)
  endif()
  add_test(NAME simd.avx512_fp16.shared_bmi COMMAND "${CMAKE_COMMAND}"
    "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DMODULES=${fp16_bmi_modules}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/../profile_package/check_bmi.cmake")
  set_tests_properties(simd.avx512_fp16.shared_bmi PROPERTIES LABELS "host;modules;avx512_fp16;profiles")
endif()
