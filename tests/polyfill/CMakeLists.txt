foreach(mode IN ITEMS header module)
  add_executable(native_test_polyfill_${mode} main.cc)
  target_link_libraries(native_test_polyfill_${mode} PRIVATE native::native)
  native_target_profile(native_test_polyfill_${mode} ${NATIVE_TEST_ISA})
  add_executable(native_test_polyfill_scalar_${mode} main.cc)
  target_link_libraries(native_test_polyfill_scalar_${mode} PRIVATE native::native)
  target_compile_definitions(native_test_polyfill_scalar_${mode} PRIVATE NATIVE_POLYFILL_SCALAR_ONLY=1)
  if(mode STREQUAL "header")
    target_compile_definitions(native_test_polyfill_${mode} PRIVATE NATIVE_POLYFILL_HEADERS=1)
    target_compile_definitions(native_test_polyfill_scalar_${mode} PRIVATE NATIVE_POLYFILL_HEADERS=1)
  endif()
  add_test(NAME native.polyfill.${mode} COMMAND native_test_polyfill_${mode})
  set_tests_properties(native.polyfill.${mode} PROPERTIES LABELS "host;polyfill" TIMEOUT 30)
  add_test(NAME native.polyfill.scalar_${mode} COMMAND native_test_polyfill_scalar_${mode})
  set_tests_properties(native.polyfill.scalar_${mode} PROPERTIES LABELS "host;polyfill;baseline" TIMEOUT 30)
endforeach()

# A further module boundary must retain the GMF native helpers during inlining.
add_library(native_test_polyfill_relay STATIC)
target_link_libraries(native_test_polyfill_relay PUBLIC native::native)
target_sources(native_test_polyfill_relay PUBLIC FILE_SET cxx_modules TYPE CXX_MODULES FILES relay.ccm)
add_executable(native_test_polyfill_relay_consumer relay.cc)
target_link_libraries(native_test_polyfill_relay_consumer PRIVATE native_test_polyfill_relay)
add_test(NAME native.polyfill.relay COMMAND native_test_polyfill_relay_consumer)
set_tests_properties(native.polyfill.relay PROPERTIES LABELS "host;polyfill;modules;baseline" TIMEOUT 30)
