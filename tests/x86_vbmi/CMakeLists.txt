cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_x86_vbmi_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
  return()
endif()

add_executable(native_test_x86_vbmi_metadata metadata.cc)
target_link_libraries(native_test_x86_vbmi_metadata PRIVATE native::headers)
target_compile_features(native_test_x86_vbmi_metadata PRIVATE cxx_std_26)
set_target_properties(native_test_x86_vbmi_metadata PROPERTIES
  CXX_SCAN_FOR_MODULES OFF INTERPROCEDURAL_OPTIMIZATION FALSE)
if(COMMAND native_link_clang_cl_asan)
  native_link_clang_cl_asan(native_test_x86_vbmi_metadata)
endif()
add_test(NAME native.x86.vbmi.metadata COMMAND native_test_x86_vbmi_metadata)
set_tests_properties(native.x86.vbmi.metadata PROPERTIES LABELS "host;baseline;x86;vbmi")

# Stronger compiler minima are compile-only: never execute these on the build host.
foreach(minimum IN ITEMS enabled disabled)
  set(target native_test_x86_vbmi_minimum_${minimum})
  add_library(${target} OBJECT metadata.cc)
  target_link_libraries(${target} PRIVATE native::headers)
  target_compile_features(${target} PRIVATE cxx_std_26)
  set_target_properties(${target} PROPERTIES
    CXX_SCAN_FOR_MODULES OFF INTERPROCEDURAL_OPTIMIZATION FALSE)
  set(options -mavx512vbmi)
  if(minimum STREQUAL "enabled")
    target_compile_definitions(${target} PRIVATE NATIVE_TEST_VBMI_MINIMUM=1)
  else()
    list(APPEND options -mno-avx512vbmi)
  endif()
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    foreach(option IN LISTS options)
      target_compile_options(${target} PRIVATE "/clang:${option}")
    endforeach()
  else()
    target_compile_options(${target} PRIVATE ${options})
  endif()
endforeach()
