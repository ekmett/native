cmake_minimum_required(VERSION 4.4)
project(native_types_check LANGUAGES CXX)
string(REPLACE " -TP " " " CMAKE_CXX_COMPILE_OBJECT "${CMAKE_CXX_COMPILE_OBJECT}")
set(NATIVE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NATIVE_ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(NATIVE_ENABLE_IPO OFF CACHE BOOL "" FORCE)
get_filename_component(native_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${native_root}" native EXCLUDE_FROM_ALL)
enable_testing()
foreach(kind IN ITEMS direct reexport)
  add_executable(types_${kind} main.cc peer.cc)
  target_link_libraries(types_${kind} PRIVATE native::common)
  if(kind STREQUAL "reexport")
    target_compile_definitions(types_${kind} PRIVATE NATIVE_TEST_NUMERICS=1)
  endif()
  set_target_properties(types_${kind} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(types_${kind} PRIVATE /W4 /WX /clang:-Wpedantic)
  else()
    target_compile_options(types_${kind} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  native_link_clang_cl_asan(types_${kind})
  add_test(NAME types.${kind} COMMAND types_${kind})
endforeach()
