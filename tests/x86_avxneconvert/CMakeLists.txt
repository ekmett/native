cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_x86_avxneconvert_check LANGUAGES CXX)
  if(NATIVE_AVXNE_INSTALLED)
    find_package(native CONFIG REQUIRED COMPONENTS native)
  else()
    set(NATIVE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(../.. native EXCLUDE_FROM_ALL)
  endif()
  enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(ne_objdump NAMES llvm-objdump HINTS "${compiler_directory}" REQUIRED)
function(ne_options target)
  set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /clang:-O2)
  else()
    target_compile_options(${target} PRIVATE -O2)
  endif()
endfunction()
foreach(profile RANGE 0 1)
  foreach(mode IN ITEMS raw public)
    set(target native_test_x86_avxneconvert_${mode}_${profile})
    add_library(${target} OBJECT codegen.cc)
    target_link_libraries(${target} PRIVATE native::native)
    target_compile_definitions(${target} PRIVATE NATIVE_NE_PROFILE=${profile})
    ne_options(${target})
    if(mode STREQUAL "raw")
      target_compile_definitions(${target} PRIVATE NATIVE_NE_RAW=1)
    endif()
  endforeach()
  add_test(NAME native.x86.avxneconvert.codegen_${profile} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../x86_codegen_pairs.py" --objdump "${ne_objdump}"
    --raw "$<TARGET_OBJECTS:native_test_x86_avxneconvert_raw_${profile}>"
    --public "$<TARGET_OBJECTS:native_test_x86_avxneconvert_public_${profile}>" --expected 14
    --output "${CMAKE_CURRENT_BINARY_DIR}/paired-codegen-${profile}.txt")
  add_test(NAME native.x86.avxneconvert.instructions_${profile} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/check_codegen.py" --objdump "${ne_objdump}"
    "$<TARGET_OBJECTS:native_test_x86_avxneconvert_public_${profile}>")
endforeach()
foreach(interface RANGE 0 2)
  set(target native_test_x86_avxneconvert_constants_${interface})
  add_executable(${target} constants.cc)
  target_compile_definitions(${target} PRIVATE NATIVE_NE_INTERFACE=${interface})
  target_link_libraries(${target} PRIVATE native::native)
  ne_options(${target})
  add_test(NAME native.x86.avxneconvert.constants_${interface} COMMAND ${target})
endforeach()
add_executable(native_test_x86_avxneconvert_runtime runtime.cc)
target_link_libraries(native_test_x86_avxneconvert_runtime PRIVATE native::native)
ne_options(native_test_x86_avxneconvert_runtime)
add_test(NAME native.x86.avxneconvert.runtime COMMAND native_test_x86_avxneconvert_runtime)
set_tests_properties(native.x86.avxneconvert.runtime PROPERTIES SKIP_RETURN_CODE 77)
foreach(variant RANGE 0 13)
  foreach(kind RANGE 0 3)
    set(target native_test_x86_avxneconvert_guard_${variant}_${kind})
    if(kind EQUAL 0)
      add_library(${target} OBJECT negative.cc)
    else()
      add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
    endif()
    target_link_libraries(${target} PRIVATE native::native)
    ne_options(${target})
    target_compile_definitions(${target} PRIVATE NATIVE_NE_VARIANT=${variant} NATIVE_NE_NEGATIVE=${kind})
    if(kind GREATER 0)
      set(diagnostic "requires target feature")
      if(kind EQUAL 3)
        set(diagnostic "call to consteval function.*is not a constant expression|implicit instantiation of undefined template")
      elseif(kind EQUAL 1)
        set(diagnostic "call to consteval function.*is not a constant expression")
      endif()
      add_test(NAME native.x86.avxneconvert.reject_${variant}_${kind} COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../source_targets/check_negative.py"
        --cmake "${CMAKE_COMMAND}" --build "${CMAKE_BINARY_DIR}" --target "${target}"
        --config "$<CONFIG>" --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${variant}-${kind}.log"
        --diagnostic "${diagnostic}")
      set_tests_properties(native.x86.avxneconvert.reject_${variant}_${kind} PROPERTIES RUN_SERIAL TRUE)
    endif()
  endforeach()
endforeach()
foreach(kind RANGE 4 7)
  set(target native_test_x86_avxneconvert_extra_guard_${kind})
  add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
  target_link_libraries(${target} PRIVATE native::native)
  ne_options(${target})
  target_compile_definitions(${target} PRIVATE NATIVE_NE_VARIANT=4 NATIVE_NE_NEGATIVE=${kind})
  set(diagnostic "call to consteval function.*is not a constant expression")
  if(kind EQUAL 4)
    set(diagnostic "one-past-the-end|outside the bounds")
  elseif(kind GREATER_EQUAL 5)
    set(diagnostic "constraints not satisfied|no matching function")
  endif()
  add_test(NAME native.x86.avxneconvert.reject_extra_${kind} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../source_targets/check_negative.py"
    --cmake "${CMAKE_COMMAND}" --build "${CMAKE_BINARY_DIR}" --target "${target}"
    --config "$<CONFIG>" --log "${CMAKE_CURRENT_BINARY_DIR}/reject-extra-${kind}.log"
    --diagnostic "${diagnostic}")
  set_tests_properties(native.x86.avxneconvert.reject_extra_${kind} PROPERTIES RUN_SERIAL TRUE)
endforeach()
