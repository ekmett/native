cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_x86_memory_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
  return()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(native_memory_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(NATIVE_MEMORY_OBJDUMP NAMES llvm-objdump HINTS "${native_memory_compiler_dir}" REQUIRED)

function(native_memory_options target)
  target_link_libraries(${target} PRIVATE native::native)
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /O2 /clang:-fconstexpr-steps=8000000)
  else()
    target_compile_options(${target} PRIVATE -O2 -fconstexpr-steps=8000000)
  endif()
endfunction()

foreach(interface RANGE 1 2)
  set(target native_test_x86_memory_${interface})
  add_executable(${target} main.cc)
  native_memory_options(${target})
  target_compile_definitions(${target} PRIVATE NATIVE_TEST_INTERFACE=${interface})
  if(COMMAND native_link_clang_cl_asan)
    native_link_clang_cl_asan(${target})
  endif()
  add_test(NAME native.x86.memory.${interface} COMMAND ${target})
  set_tests_properties(native.x86.memory.${interface} PROPERTIES SKIP_RETURN_CODE 77 LABELS "host;modules;x86;memory")
endforeach()

foreach(profile IN ITEMS minimal broad)
  set(target native_test_x86_memory_codegen_${profile})
  add_library(${target} OBJECT codegen.cc)
  native_memory_options(${target})
  if(profile STREQUAL "broad")
    target_compile_definitions(${target} PRIVATE NATIVE_MEMORY_BROAD=1)
  endif()
  add_test(NAME native.x86.memory.codegen.${profile}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_codegen.py"
      --objdump "${NATIVE_MEMORY_OBJDUMP}"
      --output "${CMAKE_CURRENT_BINARY_DIR}/codegen-${profile}-$<CONFIG>.disassembly"
      $<TARGET_OBJECTS:${target}> COMMAND_EXPAND_LISTS)
  set_tests_properties(native.x86.memory.codegen.${profile} PROPERTIES LABELS "host;modules;codegen;x86;memory")
endforeach()

add_library(native_test_x86_memory_boundaries OBJECT boundaries.cc)
native_memory_options(native_test_x86_memory_boundaries)
foreach(case RANGE 1 18)
  set(target native_test_x86_memory_reject_${case})
  add_library(${target} OBJECT EXCLUDE_FROM_ALL boundaries.cc)
  native_memory_options(${target})
  target_compile_definitions(${target} PRIVATE NATIVE_MEMORY_REJECT=${case})
  if(case EQUAL 9)
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
      target_compile_options(${target} PRIVATE /clang:-mno-avx /clang:-mno-avx2 /clang:-mno-avx512f)
    else()
      target_compile_options(${target} PRIVATE -mno-avx -mno-avx2 -mno-avx512f)
    endif()
  endif()
  add_test(NAME native.x86.memory.reject.${case}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_negative.py"
      --cmake "${CMAKE_COMMAND}" --build "${CMAKE_BINARY_DIR}" --config "$<CONFIG>"
      --case ${case} --log "${CMAKE_CURRENT_BINARY_DIR}/reject-${case}-$<CONFIG>.log")
  set_tests_properties(native.x86.memory.reject.${case} PROPERTIES RUN_SERIAL TRUE LABELS "host;modules;negative;x86;memory")
endforeach()
