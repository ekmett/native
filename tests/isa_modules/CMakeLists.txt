cmake_minimum_required(VERSION 3.30)
project(simd_isa_module_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)
# clang-cl recognizes .cc/.ccm; the redundant CMake -TP causes an unused-argument warning.
string(REPLACE " -TP " " " CMAKE_CXX_COMPILE_OBJECT "${CMAKE_CXX_COMPILE_OBJECT}")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(/W4 /WX /O2)
option(PROBE_THIN_LTO "Build ISA objects and consumers as ThinLTO; keep dispatcher native" OFF)
include("${CMAKE_CURRENT_LIST_DIR}/../../etc/cmake/simdProfile.cmake")
add_library(cpuid_provider OBJECT)
target_sources(cpuid_provider PUBLIC FILE_SET CXX_MODULES
  BASE_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../src"
  FILES "${CMAKE_CURRENT_LIST_DIR}/../../src/simd.cpu.x86.ccm")
target_include_directories(cpuid_provider PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
set_property(TARGET cpuid_provider PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
# The generic engine has one baseline object; CMake may synthesize consumer BMIs.
add_library(generic_engine OBJECT)
target_sources(generic_engine PUBLIC FILE_SET CXX_MODULES FILES fixture.engine.ccm)
set_property(TARGET generic_engine PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
foreach(backend avx2 avx512)
  add_library(${backend} OBJECT ${backend}.cc bridge_${backend}.cc)
  target_sources(${backend} PUBLIC FILE_SET CXX_MODULES FILES simd.${backend}.ccm)
  target_link_libraries(${backend} PRIVATE generic_engine)
  simd_target_profile(${backend} ${backend})
  if(PROBE_THIN_LTO)
    target_compile_options(${backend} PRIVATE /clang:-flto=thin)
  endif()
endforeach()
add_library(simd STATIC $<TARGET_OBJECTS:avx2> $<TARGET_OBJECTS:avx512> $<TARGET_OBJECTS:generic_engine>)
target_link_libraries(simd PUBLIC avx2 avx512 generic_engine)
add_executable(dispatcher dispatcher.cc)
set_property(TARGET dispatcher PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
# This TU deliberately stays native baseline COFF even when ISA objects use ThinLTO.
target_link_libraries(dispatcher PRIVATE simd cpuid_provider)
function(consumer name isa)
  add_executable(${name} EXCLUDE_FROM_ALL consumer.cc)
  if(NOT isa STREQUAL "baseline")
    simd_target_profile(${name} ${isa})
  endif()
  if(PROBE_THIN_LTO)
    target_compile_options(${name} PRIVATE /clang:-flto=thin)
  endif()
  foreach(backend IN LISTS ARGN)
    target_link_libraries(${name} PRIVATE simd)
    string(TOUPPER ${backend} upper)
    target_compile_definitions(${name} PRIVATE USE_${upper}=1)
  endforeach()
endfunction()
consumer(only_avx2 avx2 avx2)
consumer(only_avx512 avx512 avx512)
consumer(both_avx512 avx512 avx2 avx512)
consumer(both_avx2 avx2 avx2 avx512)
consumer(both_baseline baseline avx2 avx512)
