cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(simd_bf16_consumer LANGUAGES CXX)
  find_package(simd CONFIG REQUIRED COMPONENTS simd)
  enable_testing()
  set(bf16_standalone TRUE)
endif()
if(NOT TARGET simd::avx512_bf16)
  if(bf16_standalone)
    message(FATAL_ERROR "This fixture requires a package configured with AVX512_BF16.")
  endif()
  return()
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(simd_bf16_kernel OBJECT kernel.cc)
target_link_libraries(simd_bf16_kernel PRIVATE simd::minimal simd::avx512_bf16)
simd_target_profile(simd_bf16_kernel AVX512_BF16)
target_include_directories(simd_bf16_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
set_target_properties(simd_bf16_kernel PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_library(simd_bf16_omnibus OBJECT omnibus.cc)
target_link_libraries(simd_bf16_omnibus PRIVATE simd::simd simd::minimal)
simd_target_omnibus(simd_bf16_omnibus)
target_precompile_headers(simd_bf16_omnibus PRIVATE <array> <concepts> <cstddef> <type_traits>)
add_executable(simd_test_bf16 main.cc $<TARGET_OBJECTS:simd_bf16_kernel>)
target_link_libraries(simd_test_bf16 PRIVATE simd::minimal simd::avx512_bf16)
set_target_properties(simd_test_bf16 PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_dependencies(simd_test_bf16 simd_bf16_omnibus)
if(COMMAND simd_link_clang_cl_asan)
  simd_link_clang_cl_asan(simd_test_bf16)
endif()
if(TARGET simd::avx2 AND TARGET simd::avx512)
  add_library(simd_bf16_identity OBJECT identity.cc)
  target_link_libraries(simd_bf16_identity PRIVATE simd::minimal simd::avx2 simd::avx512 simd::avx512_bf16)
  simd_target_profile(simd_bf16_identity AVX512_BF16)
  add_dependencies(simd_test_bf16 simd_bf16_identity)
endif()
foreach(mode IN ITEMS none native)
  add_test(NAME simd.bf16.${mode} COMMAND simd_test_bf16 ${mode})
  set_tests_properties(simd.bf16.${mode} PROPERTIES SKIP_RETURN_CODE 77
    LABELS "host;modules;bf16;exact" TIMEOUT 180)
endforeach()
get_filename_component(bf16_compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(SIMD_BF16_OBJDUMP NAMES llvm-objdump HINTS "${bf16_compiler_directory}")
get_target_property(bf16_usage_options simd::minimal INTERFACE_COMPILE_OPTIONS)
# Sanitizer instrumentation deliberately introduces calls; inspect ordinary objects.
if(SIMD_BF16_OBJDUMP AND NOT bf16_usage_options MATCHES "sanitize=address")
  add_test(NAME simd.bf16.codegen COMMAND "${CMAKE_COMMAND}"
    "-DOBJDUMP=${SIMD_BF16_OBJDUMP}" "-DOBJECT=$<TARGET_OBJECTS:simd_bf16_kernel>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/codegen.cmake")
  set_tests_properties(simd.bf16.codegen PROPERTIES LABELS "host;modules;bf16;codegen")
endif()
if(bf16_standalone AND CMAKE_GENERATOR STREQUAL "Ninja")
  add_test(NAME simd.bf16.shared_bmi COMMAND "${CMAKE_COMMAND}"
    "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DMODULES=simd.scalar;simd.wide;simd.numerics;simd.cpuid;simd.avx512_bf16"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/../profile_package/check_bmi.cmake")
  set_tests_properties(simd.bf16.shared_bmi PROPERTIES LABELS "host;modules;bf16;profiles")
endif()
