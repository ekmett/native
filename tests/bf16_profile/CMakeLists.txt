cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_bf16_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
  set(bf16_standalone TRUE)
endif()
if(NOT TARGET native::avx512_bf16)
  if(bf16_standalone)
    message(FATAL_ERROR "This fixture requires a package for a host supporting AVX512_BF16.")
  endif()
  return()
endif()
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT "AVX512_BF16" IN_LIST NATIVE_PROFILES)
  return()
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(native_bf16_kernel OBJECT kernel.cc)
target_link_libraries(native_bf16_kernel PRIVATE native::minimal native::avx512_bf16)
native_target_profile(native_bf16_kernel AVX512_BF16)
target_include_directories(native_bf16_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
set_target_properties(native_bf16_kernel PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_library(native_bf16_omnibus OBJECT omnibus.cc)
target_link_libraries(native_bf16_omnibus PRIVATE native::native native::minimal)
native_target_omnibus(native_bf16_omnibus)
target_precompile_headers(native_bf16_omnibus PRIVATE <array> <concepts> <cstddef> <type_traits>)
add_executable(native_test_bf16 main.cc $<TARGET_OBJECTS:native_bf16_kernel>)
target_link_libraries(native_test_bf16 PRIVATE native::minimal native::avx512_bf16)
set_target_properties(native_test_bf16 PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
add_dependencies(native_test_bf16 native_bf16_omnibus)
if(COMMAND native_link_clang_cl_asan)
  native_link_clang_cl_asan(native_test_bf16)
endif()
if(TARGET native::avx2 AND TARGET native::avx512)
  add_library(native_bf16_identity OBJECT identity.cc)
  target_link_libraries(native_bf16_identity PRIVATE native::minimal native::avx2 native::avx512 native::avx512_bf16)
  native_target_profile(native_bf16_identity AVX512_BF16)
  add_dependencies(native_test_bf16 native_bf16_identity)
endif()
foreach(mode IN ITEMS none native)
  add_test(NAME native.bf16.${mode} COMMAND native_test_bf16 ${mode})
  set_tests_properties(native.bf16.${mode} PROPERTIES SKIP_RETURN_CODE 77
    LABELS "host;modules;bf16;exact" TIMEOUT 180)
endforeach()
get_filename_component(bf16_compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(NATIVE_BF16_OBJDUMP NAMES llvm-objdump HINTS "${bf16_compiler_directory}")
get_target_property(bf16_usage_options native::minimal INTERFACE_COMPILE_OPTIONS)
# Sanitizer instrumentation deliberately introduces calls; inspect ordinary objects.
if(NATIVE_BF16_OBJDUMP AND NOT bf16_usage_options MATCHES "sanitize=address")
  add_test(NAME native.bf16.codegen COMMAND "${CMAKE_COMMAND}"
    "-DOBJDUMP=${NATIVE_BF16_OBJDUMP}" "-DOBJECT=$<TARGET_OBJECTS:native_bf16_kernel>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/codegen.cmake")
  set_tests_properties(native.bf16.codegen PROPERTIES LABELS "host;modules;bf16;codegen")
endif()
if(bf16_standalone AND CMAKE_GENERATOR STREQUAL "Ninja")
  add_test(NAME native.bf16.shared_bmi COMMAND "${CMAKE_COMMAND}"
    "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DMODULES=native.scalar;native.wide;native.simd;native.numerics;native.x86.features;native.features;native"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/../profile_package/check_bmi.cmake")
  set_tests_properties(native.bf16.shared_bmi PROPERTIES LABELS "host;modules;bf16;profiles")
endif()
