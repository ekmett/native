cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(native_arm_crc_consumer LANGUAGES CXX)
  find_package(native CONFIG REQUIRED COMPONENTS native)
  enable_testing()
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  return()
endif()
set(interfaces header module arm hub)
set(index 0)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(interfaces module arm hub)
  set(index 1)
endif()
foreach(interface IN LISTS interfaces)
  set(target native_test_arm_crc_${interface})
  add_executable(${target} main.cc)
  target_compile_definitions(${target} PRIVATE NATIVE_TEST_INTERFACE=${index})
  if(interface STREQUAL "hub")
    target_link_libraries(${target} PRIVATE native::native)
  else()
    target_link_libraries(${target} PRIVATE native::minimal)
  endif()
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(COMMAND native_link_clang_cl_asan)
    native_link_clang_cl_asan(${target})
  endif()
  add_test(NAME native.arm.crc.${interface} COMMAND ${target})
  set_tests_properties(native.arm.crc.${interface} PROPERTIES SKIP_RETURN_CODE 77 LABELS "host;arm;crc;modules")
  math(EXPR index "${index}+1")
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  return()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(crc_objdump NAMES llvm-objdump HINTS "${compiler_dir}" REQUIRED)
function(crc_baseline target)
  target_link_libraries(${target} PRIVATE native::headers)
  target_compile_features(${target} PRIVATE cxx_std_26)
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES OFF INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /O2 /clang:-mcpu=generic /clang:-march=armv8-a+nocrc)
  else()
    target_compile_options(${target} PRIVATE -O2 -mcpu=generic -march=armv8-a+nocrc)
  endif()
endfunction()
add_library(native_arm_crc_codegen OBJECT codegen.cc)
crc_baseline(native_arm_crc_codegen)
add_test(NAME native.arm.crc.codegen COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_codegen.py"
  --objdump "${crc_objdump}" --output "${CMAKE_CURRENT_BINARY_DIR}/crc-$<CONFIG>.disassembly"
  $<TARGET_OBJECTS:native_arm_crc_codegen> COMMAND_EXPAND_LISTS)
foreach(kind RANGE 0 3)
  set(target native_arm_crc_negative_${kind})
  add_library(${target} OBJECT EXCLUDE_FROM_ALL negative.cc)
  target_compile_definitions(${target} PRIVATE TEST_KIND=${kind})
  crc_baseline(${target})
  if(kind EQUAL 1)
    set(diagnostic "requires target feature|target specific option mismatch")
  else()
    set(diagnostic "deleted function|no matching function")
  endif()
  add_test(NAME native.arm.crc.reject_${kind} COMMAND "${Python3_EXECUTABLE}"
    "${PROJECT_SOURCE_DIR}/tests/source_targets/check_negative.py" --cmake "${CMAKE_COMMAND}"
    --build "${CMAKE_BINARY_DIR}" --target "${target}" --config "$<CONFIG>"
    --diagnostic "${diagnostic}" --log "${CMAKE_CURRENT_BINARY_DIR}/negative-${kind}.log")
  set_tests_properties(native.arm.crc.reject_${kind} PROPERTIES RUN_SERIAL TRUE TIMEOUT 180 LABELS "host;arm;crc;negative")
endforeach()
